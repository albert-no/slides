\section{Probability Review}
Let $\Omega$ be the {\it sample space}, which is the set of all possible outcomes.
The event $E$ is a (measurable) subset of the sample space,
and the set of events $\cF$ is a collection of subsets (events) of $\Omega$.
The {\it probability measure} is a map $P: \cF \rightarrow [0, 1]$.

\begin{example}[Fair Coin]
    We can set $\Omega = [H, T]$, and $\cF = \{\emptyset, \{H\}, \{T\}, \{H, T\}\}$.
    The probability measure is given by
    \begin{align}
        \Pr{\emptyset} = 0, \Pr{\{H\}} = \Pr{\{T\}} = 1/2, \Pr{\{H, T\}} = 1.
    \end{align}
\end{example}

\begin{example}[$5\times 5$ 0-1 image]
    An image $5\times 5$ is simply a 25-dimensional binary vector.
    In this case, $\Omega = \{0, 1\}^{5\times 5}$ where $|\Omega| = 2^{25}$.
    Clearly $|\cF| = 2^{2^{25}}$.

    Note that $[0, 1]$ is an interval (set of real numbers) and $\{0, 1\}$ is a set of size 2.

    \centering
    \includegraphics[width=0.4\textwidth]{images/binary_image.png}
    \caption{$5\times5$ binary image}
\end{figure}
\end{example}

\begin{example}[MNIST]
    MNIST image is a black-white image $28\times 28$, where the intensity of each pixel is from 0 to 255.
    $\Omega = \{0, 1, \dots, 255\}^{28\times 28}$.
    The probability on $\Omega$ is not uniform (of course).

    \caption{MNIST}

\end{example}

\begin{example}[CIFAR10]
    CIFAR10 image is $32\times 32$ RGB color image where the intensity of each pixel ranges from 0 to 255.
    $\Omega = \{0, 1, \dots, 255\}^{3 \times 32\times 32}$

\end{example}

\section{Random Variable}
Random variable is a map from sample space to real number, that is, $X:\Omega\rightarrow \fR$.
In other words, convert a sample to a number.

\begin{example}[$5\times 5$ 0-1 image]
    In case of $\Omega = \{0, 1\} ^{5\times 5}$, a random variable $X$ can be number of 1's.    
\end{example}

\begin{example}[7 coin tosses]
    In case of $\Omega = \{H, T\} ^{7}$, a random variable $X$ can be the number of heads followed by a tail.
    For example, $X(HTHTHHH)=2$.
\end{example}

Probability mass function (pmf) of random variable $X$ is given by
\begin{align}
    p_X(x) = \Pr{X=x}
\end{align}
Note that we use uppercase to emphasize the $X$ is random variable,
while we use lowercase $x$ to denote realization, i.e., $\Pr{X=x}$.

We let $\cX$ be the set of possible values of the random variable $X$.
Note that the probability mass function completely characterizes the random variable.

\begin{definition}
    The expectation of random variable $X$ is defined by
    \begin{align}
        \Expec{X} = \sum_{x\in\cX} x \cdot p_X(x).
    \end{align}
\end{definition}
\begin{exercise}
    For any function $f:\cX\rightarrow \fR$, show that
    \begin{align}
        \Expec{f(X)} = \sum_{x\in\cX} f(x)\cdot p_X(x).
    \end{align}
    Hint: You can set a new random variable $Y= f(X)$.
\end{exercise}

\section{Entropy}
Let $\cX= \{0, 1, \dots, M-1\}$, and $X$ be a discrete random variable with pmf $p_X(x)$. 

\begin{definition}
    ``Surprisal'' of $X$ at $x$ is given by
    \begin{align}
        S(x) = \log \frac{1}{p_X(x)} = -\log p_X(x).
    \end{align}
\end{definition}
This is a natural definition, since if rare events occur (where $p_X(x)$ is small), then it is ``surprising''.

\begin{example}
    Let $X$ be an indicator where $X=1$ is a serious earthquake and $X=0$ otherwise.   
    $p_X(1)$ is small,
\end{example}


The entropy would be an expected suprisal of random variable $X$.
\begin{definition}
    The entropy of the random variable $X$ is given by \begin{align}
        H(X) =& \Expec{S(X)}\\
        =& \Expec{-\log p_X(X)}\\
        =& \sum_{x=1}^M p_X(x) \log \frac{1}{p_X(x)}.
    \end{align}
    Note that we follow the convention of $0\cdot \log 0 = 0$, and $\log$ is base 2.
\end{definition}

\begin{example}
    Let $X\in\{0, 1\}$ be a binary random variable where $p_X(0) = 0.01$ and $p_X(1) = 0.99$, then
    \begin{align}
        H(X) = 0.01 \log \frac{1}{0.01} + 0.99 \log \frac{1}{0.99}.
    \end{align}
\end{example}

\begin{exercise}
    Find the entropy of fair dice.
\end{exercise}

Entropy is ``measure of uncertainty (randomness)'' and also ``amount of information.''
Unit of entropy is often ``bit''s.

\begin{example}
    Consider two random variables $\cX = \{0, 1\}$ with $p_X(0) = p_X(1) = 1/2$
    and $\cY = \{0, 1\}$ with $p_Y(0) = 0.99, p_Y(1) = 0.01$.
    Which random variable is more random? (more uncertain?)
\end{example}

\begin{example}[Guessing game]
    Alice picks a number from 0 to 7, and Bob wants to guess it.
    Bob can ask ``yes or no'' questions, and which of the following questions is more informative?
    \begin{enumerate} [(Q1)]
        \item Is your number is in $\{0, 1, 2, 3\}$? (probability of yes is $1/2$)
        \item Is your number is in $\{3, 6\}$? (probability of yes is $1/4$)
    \end{enumerate}
    It is clear that Q1 is more informative question (which is because it has higher entropy).
\end{example}

\begin{example}[Dice]
    Consider the following two dices
    \begin{enumerate}[(D1)]
        \item probability assignment is $(1/9, 1/9, 1/9, 2/9, 2/9, 2/9)$
        \item probability assignment is $(1/2. 1/4, 1/8, 1/16, 1/32, 1/32)$
    \end{enumerate}
    Which is more uncertain?
\end{example}

\begin{example}[Classifier]
    You have a machine learning algorithm $f$ that classifies the handwritten digits from 0 to 4.
    Like many other classifiers, your algorithm $f$ also outputs the soft label, i.e.,
    \begin{align}
        f(x) = (0.4, 0.4, 0.2, 0, 0)
    \end{align}
    implies that your algorithm thinks $x$ is a digit ``0'' with probability 0.4, and $x$ is digit ``2'' with probability 0.2.
    For two images $x_1$ and $x_2$, suppose that the outputs of your algorithm are
    \begin{align}
        f(x_1) =& (0.6, 0.1, 0.1, 0.1, 0.1)\\
        f(x_2) =& (0.4, 0.4, 0.2, 0.0, 0.0)
    \end{align}
    which sample is harder to classify?
\end{example}

\begin{remark}
    The above is called soft guessing (like the weather forecast).
\end{remark}

\begin{remark}
    The above example can be interpreted as a ``data valuation.''
Data valuation asks the question ``which sample is the most helpful while training?''
    It is often the case where you may want to find the most challenging sample to classify (so that your model can learn how to classify those challenging samples).
\end{remark}

\begin{example}[Reinforcement Learning]
    Suppose that you have strategy options A, B, C, D, E, for example, in the GO game.
    Your current belief is ``A is the best strategy''.
    In this round of the game, you can choose ``A'' to ``exploit'' (which won an immediate reward);
however, you may want to ``explore'' other options.
    In reinforcement learning, balancing exploration and exploitation is important, 
    and we often pick strategy in a probabilistic manner.
    You may want to assign a higher probability to ``A'', but you may also want to reserve the probability for other exploration options.
    Suppose you have two distribution 
    \begin{enumerate}[(S1)]
        \item $(\Pr{A}, \Pr{B},\Pr{C},\Pr{D},\Pr{E}) = (0.6, 0.1, 0.1, 0.1, 0.1)$ 
        \item $(\Pr{A}, \Pr{B},\Pr{C},\Pr{D},\Pr{E}) = (0.4, 0.4, 0.2, 0.0, 0.0)$ 
    \end{enumerate}
    Which distribution is more ``explorative''?
    Entropy can be a measure of ``explorativeness''.
\end{example}

\begin{exercise}
    Let $X$ be a geometric distribution with $p=1/2$. I.e.,
    \begin{align}
        p_X(x) = \frac{1}{2^x}
    \end{align}
    for $x\geq 1$.
    What is the entropy of $X$?
\end{exercise}

\section{Properties of Entropy}
\begin{theorem}[Non-negativity]
    Entropy is non-negative, and it is 0 if and only if $X$ is deterministic.
\end{theorem}
\begin{proof}
    Since the probability is always $0\leq p_X(x)\leq 1$, we have
    \begin{align}
        H(X) =& \Expec{\log \frac{1}{p_{X}(X)}}\\
        \geq & \Expec{\log 1}\\
        =& 0.
    \end{align}
    The equality holds if and only if $p_X(x)\log p_X(x) = 0$ for all $x$,
    which is true only if $X$ is deterministic.
\end{proof}

\begin{remark}
    $H(X)=0$ means $X$ has no randomness, and therefore deterministic (vice versa).
\end{remark}

\begin{question}
    Which distribution maximizes the entropy?
\end{question}

\begin{definition}[Convexity (Concavity)]
A function $f:\cX \rightarrow \fR$ is convex if
\begin{align}
    \lambda f(x) + (1-\lambda) f(y) \geq f(\lambda x + (1-\lambda) y)
\end{align}
for all $x, y$ and $0\leq \lambda \leq 1$.
In other words, the weighted average of $f$ is smaller than or equal to $f$ on the weighted average input.
The function $f$ is concave if 
\begin{align}
    \lambda f(x) + (1-\lambda) f(y) \leq f(\lambda x + (1-\lambda) y)
\end{align}
for all $x, y$ and $0\leq \lambda \leq 1$.
\end{definition}

\begin{figure}
    \centering
    \includegraphics[width=0.7\textwidth]{images/jensen.png}
    \caption{Jensen's inequality}
\end{figure}

\begin{exercise}
    Let $X$ be a Bernoulli random variable with parameter $p$.
    Then, we define a binary entropy function $h_2(p) = H(X) = -p\log p - (1-p)\log (1-p)$.
    Show that $h_2(p)$ is concave.
\end{exercise}

\begin{theorem}[Jensen's Inequality]
Let $f$ be a concave function, then
\begin{align}
    \Expec{f(X)} \leq f(\Expec{X}).
\end{align}
\end{theorem}
Then, Jensen's inequality is a generalized version of the definition of concavity,
where $\Expec{f(X)}$ is weighted average of $f$ and $f(\Expec{X})$ is a function value at weighted average input.
For strictly concave function $f$, Jensen's inequality holds with equality if and only if $f(x)$ is constant.

\begin{theorem}[Maximum Entropy]
For a random variable $X\in\cX$ where $|\cX|=M$, we have
\begin{align}
    H(X) \leq \log |\cX| = \log M.
\end{align}
\end{theorem}
\begin{proof}
Since $\log$ is a concave function, by Jensen's inequality
\begin{align}
    H(X) =& \Expec{\log \frac{1}{p_X(X)}}\\
    \leq& \log \Expec{\frac{1}{p_X(X)}}\\
    =& \log \sum_{x=1}^M p_X(x) \frac{1}{p_{X}(x)}\\
    =& \log M.
\end{align}
\end{proof}

Equality of the above inequality holds if and only if $\frac{1}{p_X(x)}$ is constant,
which implies $X$ is uniform on $\{1,\dots,M\}$, i.e., $p_X(x) = 1/M$ for all $x$.

\begin{remark}
    Why are we using ``Surprisal'' as a measure of uncertainty? Why are we using $\log$?
    Information theory has an answer for this question (axiomatic approach, etc.), but will not discuss here.
    However, we will see that the above definition of entropy has a lot of good properties.
    Note that there are other measures of information, including R\`enyi entropy, etc.
\end{remark}

\begin{theorem}[Mismatch]\label{thm:mismatch}
For ``wrong'' probability mass function $Q(x)$,
\begin{align}
    H(X) \leq \Expec{\log \frac{1}{Q(X)}} = \sum_{x} p_X(x) \log \frac{1}{Q(x)}.\label{eq:wrong pmf}
\end{align}
\end{theorem}
\begin{example}
    Let $X$ be the same binary distribution $P_X(1) = 0.01$ and $P_X(2) = 0.99$ where $Q(1) = 0.1$ and $Q(2) = 0.9$, then the right hand side would be
    \begin{align}
        \Expec{\log \frac{1}{Q(X)}} = 0.01 \log \frac{1}{0.1} + 0.99 \log\frac{1}{0.9}.
    \end{align}
\end{example}
\begin{proof}
    Since $\log$ is concave function, by Jensen's inequality~\cite{jensen1906fonctions},
    \begin{align}
        H(X) - \Expec{\log \frac{1}{Q(X)}} =& \Expec{\log \frac{Q(X)}{p_X(X)}}\\
        \leq& \log \Expec{\frac{Q(X)}{p_X(X)}}\\
        =& \log \sum_x p_X(x)\frac{Q(x)}{p_X(x)}\\
        =& \log \sum_x Q(x)\\
        =& \log 1 =0.
    \end{align}
    Equality holds if and only if $\frac{Q(x)}{p_X(x)}$ is constant, which implies $Q = p_X$.
\end{proof}

\begin{definition}[Kullback-Leibler (KL) Divergence]
For two distributions $P$ and $Q$ on $\cX$, the KL divergence (or relative entropy) is given by
\begin{align}
    D(P \|Q) =  \sum_{x\in\cX} P(x) \log \frac{P(x)}{Q(x)}.
\end{align}
\end{definition}
\begin{remark}
    From the above proof, we have
    \begin{align}
        \Expec{\log \frac{p_X(X)}{Q(X)}} = \sum_{x} p_X(x) \log\frac{p_X(x)}{Q(x)}\geq 0
    \end{align}
    which implies
    \begin{align}
        D(p_X\|Q) \geq 0.
    \end{align}
    In other words, the KL divergence is always non-negative and 0 if and only if $p_X=Q$.
\end{remark}

    Prove Theorem~\ref{thm:mismatch} rigorously.    
\end{exercise}

\begin{definition}
    The right-hand side of \eqref{eq:wrong pmf} is called ``cross entropy,''
    where we happen to use the wrong pmf $Q(x)$ instead of the true pmf $P_X(x)$.
\end{definition}

\begin{example}[Gambling]
    Consider a binary game where you can bet `red' or `black'.
    The probability of `red' is $p$ and the probability of `black' is $1-p$ where $p<1/2$.
    The odds are $2$ for both case.
    If you have a budget $B$, what would be the best strategy.
    Let $S$ be your budget after the bet, when you bet $qB$ on `red' and $(1-q)B$ on `black'.
    We have
    \begin{align}
        \Expec{S} = p\cdot 2qB + (1-p)\cdot 2(1-q)B
    \end{align}
    which is maximized when $q=0$, in other words, bet all your money on `black'.
    However this strategy is suboptimal if you repeat the same strategy many times, 
    since the probability of losing at least once is high and your budget becomes 0.
    One way of prevent losing all your money is maximizing $\Expec{\log S}$ since $\log 0 = -\infty$.
    Informally, it forces the strategy that the probability of losing all your money to be zero.
    In this case,
    \begin{align}
        \Expec{\log S} = p\log 2qB + (1-p) \log 2(1-q)B = \log 2B - (p\log \frac{1}{q} + (1-p)\log\frac{1}{1-q}),
    \end{align}
    which is maximized when $p=q$.
    I.e., the best strategy is bet $p$ portion to `red' and $1-p$ fraction to `black'.
    Note that the `cost of mismatch', the difference between your strategy ($q,1-q$) and the best strategy ($p, 1-p$), is $D((p, 1-p)\|(q, 1-q))$.
    This implies that the difference is proportional to the ``distance'' between two distributions $(p, 1-p)$ and $(q, 1-q)$,
    where the distance is measured by KL divergence.

    Also, let $W = \log 2 - (p\log \frac{1}{q} + (1-p)\log\frac{1}{1-q})$, then roughly speaking we have
    \begin{align}
        S \approx B \times 2^W.
    \end{align}
    And if you repeat the same strategy $n$ times, then your budget after $n$ games would be roughly
    \begin{align}
        S_n \approx B \times 2^{nW}.
    \end{align}
    The $W$ is often called doubling rate.
\end{example}

\begin{example}[Horse Racing]
    Similarly, consider the horse racing with $M$ horses, where the proability of $i$-th horse wins the game is $p_X(i)$.
    The odds are $M$ for all horses.
    Your strategy is divide your budget into $(Q(1)B, \dots, Q(M) B)$ to each horse.
    Similar to the previous example, let $S$ be the budget after the game, then
    \begin{align}
        \Expec{\log S} =& \sum_{i=1}^M p_X(i) \log (M Q(i)B)\\
        =& \log B + \log M - \sum_{i=1} p_X(i) \log\frac{1}{Q(i)}
    \end{align}
    which is maximized when $Q=p_X$.
    The cost of mismatch is also $D(p_X \| Q)$, the distance between $p_X$ and $Q$.
\end{example}

\section{Cross Entropy Loss} \label{sec:intro-cross}
\begin{figure}[h]
    \centering
    \includegraphics[width=0.3\textwidth]{images/classification.png}
    \caption{Classifier $f$}
\end{figure}

Consider a classification problem where data input $x\in\cX$ is from the set of possible inputs (such as $\fR^n$)
and the corresponding label is $y\in [M] = \{1, 2, \dots, M\}$.
A classifier $f:\cX \rightarrow \cM$ produces the probability distribution $f(x) = (f(x)_1, \dots, f(x)_M)$ 
where $f(i)$ indicates the ``the probability of $x$ corresponds to label $i$.''
Note that $\cM$ is a set of probability measures on set of labels.
The cross entropy is defined by
\begin{align}
    \ell(f(x), y) = \log \frac{1}{f(x)_y}
\end{align}


In one hand, it is weird looking function:
1) it is not clear why we are using $\log$ function,
2) it is not clear why we are using reciprocal.
Of course, the definition make sense, since ``if your model assigns the higher value to the true label, the loss value is lower.''
However, such property holds for any decreasing function, it does not have to be $\log 1/f(y)$.

Recall that KL divergence is defined by 
\begin{align*}
    D(P\| Q) = \sum_{x\in[M]} P(x) \log \frac{P(x)}{Q(x)}.
\end{align*}
It measures a ``distance'' between two probability distribution\footnote{strictly speaking
it is not a distance since it is not symmetric.}.
Let $y^{(o)}\in\fR^M$ is an one-hot vector represents ground truth, i.e.,
\begin{align}
    y^{(o)}_i = \begin{cases}
        1 &\mbox{ if $y=i$}\\
        0 &\mbox{ otherwise}
    \end{cases}    
\end{align}
The one-hot vector $y^{(o)}$ can be viewed as a ground truth distribution of input $x$,
which is deterministically pointing the true label $y$.
Then, it is natural to consider the ``distance'' between the true distribution $y^{(o)}$ and the output of classifier $f(x)$
to measure the performance.

The KL divergence between $f(x)$ and $y^{(o)}$ is
\begin{align}
    D(y^{(o)}\| f(x)) 
    =&  \sum_{i=1}^M y^{(o)}_i \log \frac{f(x)_i}{y^{(o)}_i}\\
    =& \log \frac{1}{f(x)_y}\\ 
    =& \ell(f(x),y).
\end{align}
In other words, the KL divergence is equivalent to cross entropy,
which justifies the usage of the cross entropy loss function.

However, the concern remains.
The KL divergence is also weird looking.
Why are we measuring ``distance'' between distributions in such way?
Is there another way of measuring ``distance''?
Why is it called divergence? instead of ``distance''?

\begin{remark}
    We will see later, KL divergence plays a crucial role in Generative Adversarial Networks (GAN).
\end{remark}



For general distribution (other than one-hot distribution), we have
\begin{align}
    D(p_X\|Q) =& \sum_{x} p_X(x) \log \frac{p_X(x)}{Q(x)}\\
    =& -\sum_{x} p_X(x) \log \frac{1}{p_X(x)} + \sum_x \log\frac{1}{Q(x)}\\
    =& -H(p_X) +\sum_x \log p_X(x)\frac{1}{Q(x)}.
\end{align}
where the second term is called cross entropy which coincides with the cross entropy loss
\begin{align}
    CE(y^{(o)}\|f(x)) = \sum_m y^{(o)}_m \log \frac{1}{f(x)_m}.
\end{align}
This is the reason why we call it ``cross entropy loss.''
Thus, even when the true label is ``soft'' distribution (instead of one-hot), we have
\begin{align}
    D(y^{(soft)} \| f(x)) = -H( y^{(soft)}) + CE(y^{(soft)} \| f(x)).
\end{align}
Since the true label is given and fixed, minimizing KL divergence is equivalent to minimizing cross entropy loss.


\begin{remark}
    We often abuse the notation $H(X) = H(p_X)$.
\end{remark}

\section{Jointly Distributed Random Variables}
Let $X\in \cX$ and $Y\in \cY$ are random variables.
The joint probability mass function (pmf) is given by
\begin{align}
    p_{X, Y} (x, y) = \Pr{X=x, Y=y}.
\end{align}
The marginal distribution can be obtained by summing joint pmf with respect to the separate random variable:
\begin{align}
    p_X(x) &= \sum_{y\in\cY} p_{X,Y}(x, y) = \sum_{y\in \cY} p_{X|Y}(x|y)p_Y(y) = \Expec{p_{X|Y}(x|Y)}
\end{align}
where the expectation is with respect to the random variable $Y$.

\begin{exercise}
    Consider jointly distributed random variables $X$ and $Y$. 
    For any functions $f:\cX\rightarrow \fR$ and $g:\cY\rightarrow \fR$, show that
    \begin{align}
        \Expec{f(X) + g(Y)} = \Expec{f(X)}  + \Expec{g(Y)}.
    \end{align}
\end{exercise}
The above property is ``linearity'' of expectation, which is true for any jointly distributed random variables.

\begin{definition}
    The random varaibles $X$ and $Y$ are independent if and only if
    \begin{align}
        p_{X,Y}(x, y) = p_X(x) p_Y(y)
    \end{align}
    for all $x\in\cX, y\in\cY$.
\end{definition}

\begin{exercise}
    Let $X$ and $Y$ be a joint random variable with pmf $p_{X, Y}(x, y)$.

\subsection{Joint Entropy}
\begin{definition}[Joint Entropy]
    For jointly distributed random variables $X_1$ and $X_2$ with joint probability mass function $P_{X_1, X_2}(x_1, x_2)$.
    The joint entropy is given by
    \begin{align}
        H(X_1, X_2) =& \Expec{\log \frac{1}{p_{X_1, X_2}(X_1, X_2)}}\\
        =& \sum_{x_1, x_2} p_{X_1, X_2} (x_1, x_2) \frac{1}{p_{X_1, X_2} (x_1, x_2)}.
    \end{align}
\end{definition}
Recall that the expectation of the joint function $f(x,y)$ is simply the sum of $p_{X_1, X_2}(x_1,x_2)f(x_1,x_2)$.
\begin{equation}
\Expec{f(X_1, X_2)}=\sum p_{X_1, X_2}(x_1,x_2)f(x_1,x_2).
\end{equation}
Note that $X_1$ and $X_2$ may have different support, i.e., $X_1\in[M] = \{1, \dots,M\}$ and $X_2 \in [N] = \{1, \dots, N\}$.


%%% Properties of Entropy
\begin{theorem}[Property of Entropy 3]
    If $X_1$ and $X_2$ are independent,
    \begin{align}
        H(X_1, X_2) = H(X_1) + H(X_2).
    \end{align}
\end{theorem}
\begin{proof}
If $X_1$ and $X_2$ are independent, then $p_{X_1, X_2} (x_1, x_2) = p_{X_1}(x_1) p_{X_2}(x_2)$ for all $x_1, x_2$.
\begin{align}
    H(X_1, X_2) =& \sum_{x_1, x_2} p_{X_1, X_2} (x_1, x_2) \log \frac{1}{p_{X_1}(x_1)p_{X_2}(x_2)}\\
    =& \sum_{x_1, x_2} p_{X_1, X_2} (x_1, x_2) \log \frac{1}{p_{X_1}(x_1)}
    + \sum_{x_1, x_2} p_{X_1, X_2} (x_1, x_2) \log \frac{1}{p_{X_2}(x_2)}\\
    =& \sum_{x_1}  \log \frac{1}{p_{X_1}(x_1)}\sum_{x_2}p_{X_1, X_2} (x_1, x_2)
    + \sum_{x_2} \log \frac{1}{p_{X_2}(x_2)}\sum_{x_1} p_{X_1, X_2} (x_1, x_2) \\
    =& \sum_{x_1}  \log \frac{1}{p_{X_1}(x_1)} p_{X_1} (x_1)
    + \sum_{x_2} \log \frac{1}{p_{X_2}(x_2)} p_{X_2} (x_2) \\
    =& H(X_1)+H(X_2).
\end{align}
\end{proof}
It is clear that if $X_1$ and $X_2$ are highly correlated,
the amount of information that we obtain from $(X_1, X_2)$ should be similar to
the amount of information from $X_1$.

\begin{exercise}
    If $H(X_1, X_2)=H(X_1) +H(X_2)$, then does this imply independence?
\end{exercise}

The Property 3 can be generalized to multi variables.
If $X_1, \dots, X_n$ are independent, i.e., $p_{X_1,\dots, X_n}(x_1,\dots, x_n) = \prod_{i=1}^n p_{X_i}(x_i)$ ,then
\begin{align}
    H(X_1,\dots, X_n) = \sum_{i=1}^n H(X_i). 
\end{align}

\begin{example}[Guessing Game]
    Recall the guessing game.
    Alice picks number $X$ (uniform at random) from $\{1, 2, \dots, 8\}$.
    Bob asks questions:
    \begin{enumerate}[Q1:]
        \item Is $X\in\{5, 6, 7, 8\}$? The random variable $Y_1=1$ if the answer is yes, and $Y_1=0$ if not.
        \item Is $X\in\{1, 2, 5, 6\}$? The random variable $Y_2=1$ if the answer is yes, and $Y_2=0$ if not.
        \item Is $X\in\{1, 3, 5, 7\}$? The random variable $Y_3=1$ if the answer is yes, and $Y_3=0$ if not.
    \end{enumerate}
    Note that $Y_1, Y_2, Y_3$ are all Bernoulli $1/2$ random variable, and therefore $H(Y_1) = H(Y_2) = H(Y_3) = 1$.
    Also, $Y_1, Y_2, Y_3$ are independent since
    \begin{align}
        p_{Y_1, Y_2, Y_3}(y_1, y_2, y_3) = 1/8 = p_{Y_1}(y_1)p_{Y_2}(y_2)p_{Y_3}(y_3)
    \end{align}
    for all $y_1, y_2, y_3$.
    Thus, $H(Y_1, Y_2, Y_3) = H(Y_1)+H(Y_2)+H(Y_3) = 3 = H(X)$.
    This implies that each question worth of 1 bit information, and the total information we can obtain after three question is 3 bits (if questions are independent).
    Since the amount of uncertainty (amount of information) in $X$ is also 3 bits, three question is enough to deterministically estimate the random variable $X$.
\end{example}

\subsection{Conditional Entropy}
In the above guessing game example, consider the Q4 where ``Is $X\in\{4, 5, 6, 7\}$?'' 
The random variable $Y_4=1$ if the answer is yes, and $Y_4=0$ if not.
It is clear that $Y_1$ and $Y_4$ are not independent.
Then, what is the amount of information you can get by asking two questions Q1 and Q4?
The fact that $Y_1$ and $Y_4$ are dependent implies that the information from $Y_4$ is known in part when you obtain $Y_1$.
Thus, $H(Y_1, Y_4)$ should be strictly smaller than $H(Y_1)+H(Y_4)$.
The question is can we formally describe the amount of additional information when you get $Y_4$ if $Y_1$ is already known?
It leads us to the definition of conditional entropy.

\begin{definition}
The conditional entropy $H(X|Y)$ is an amount of information of X given the information of Y, which can be defined as 
\begin{align}
    H(X|Y) =& \mathbb{E}[\log\frac{1}{p_\mn{X|Y}(X|Y)}]\\
    =&\sum_{}^{}p_\mn{X,Y}(x,y)\log\frac{1}{p_\mn{X|Y}(x|y)}.
\end{align}

\end{definition}
In our case, the function that we want to compute expectation is $\log\frac{1}{P_\mn{X|Y}(x,y)}$, and the probability of $x,y$ can be expressed with joint distribution $P_\mn{X,Y}(x,y)$.

We are considering random variables $X$ and $Y$ to compute the expectation.
The joint entropy is a sum of individual entropies, which becomes apparent when we consider each outcome $y$ and its respective contribution to the overall entropy.
Let's consider the case that we fix the certain condition $y$, then the expression can be written as:

\begin{equation}\label{eq:jointentropy}
H(X|Y=y)=\mathbb{E}[\log\frac{1}{p_\mn{X|Y}(X|Y=y)}]
\end{equation}

Now the only variable in the expectation term is $X$, so the equation \eqref{eq:jointentropy} can be expressed as:
\begin{equation}
\sum_{}^{}p_\mn{X|Y}(x|y)\log\frac{1}{p_\mn{X|Y}(x|y)}
\end{equation}

If we aggregate all possible instances of conditional entropy given a specific condition $y$, we can represent the overall conditional entropy of $X$ given $Y$ as:
\begin{equation}
H(X|Y)=\sum_y^{}p_Y(y)H(X|Y=y)
\end{equation}

Now, let's consider the entropies $H(X)$ and $H(Y|X)$.
$H(X)$ is an information of $X$, while $H(Y|X)$ is a ``remaining" information in $H(X,Y)$ when the information of $X$ is given.
Therefore, we expect $H(X)+H(Y|X)=H(X,Y)$, and we can prove it.

\begin{proof}
\begin{align}
     \mathbb{E}[\log\frac{1}{p_X(X)}]+ \mathbb{E}[\log\frac{1}{p_\mn{Y|X}(Y|X)}]
    & =\mathbb{E}[\log\frac{1}{p_X(X)p_\mn{Y|X}(Y|X)}]\\
    & =\mathbb{E}[\log\frac{1}{p_\mn{X,Y}(X,Y)}]\\
    & =H(X,Y).
\end{align}
\end{proof}

\begin{exercise}
    In the above guessing game, compute
    \begin{enumerate}
        \item $H(Y_2|Y_1)$
        \item $H(Y_4|Y_1)$
    \end{enumerate}
\end{exercise}

With a definition of conditional entropy, we can prove the first data processing inequality.
\begin{theorem}[Data processing inequality] Let f be deterministic function. Then,
    \begin{align}
        H(X) \geq H(f(X))
    \end{align}
\end{theorem}

\begin{proof}
    \begin{align}
        H(X,f(X))&=H(X)+H(f(X)|X)\\
        &=H(X)\\
        H(X,f(X))&=H(f(X))+H(X|f(X))\\
        &\geq H(f(X))
    \end{align}
\end{proof}
If $f$ is one-to-one and onto (therefore there exists an inverse), then $H(X) = H(f(X))$.
The above theorem implies that the amount information may not increase by post processing (applying a deterministic function).

\subsection{Mutual Information}
\textbf{Definition.} Mutual information is defined as the subtraction of the entropy and the conditional entropy.

\begin{equation}
    \label{mutual_information}
    I(X;Y) = H(X) - H(X|Y), \quad I(Y;X) = H(Y) - H(Y|X)
\end{equation}

Conditional entropy and Mutual information can be expressed in the following diagram.

\begin{figure}[!h]
    \centering
    \includegraphics[width=0.45\columnwidth]{images/scribe15_diagram.png}
    \label{fig:enter-label}
    \caption{Conditional entropy and Mutual information diagram}
\end{figure}

In the case of $I(X;X)$, it can be expressed as the below. 
As a result, $I(X;X)$ is equal to $H(X)$.

\begin{equation}
\label{x;x}
\begin{split}
    I(X;X) &= H(X) - H(X|X) \\
    &= H(X) - 0 \\
    &= H(X)
\end{split}
\end{equation}

What if $X$ and $Y$ are independent of each other?
In this case, we can prove that $I(X;Y) = 0$ through~\eqref{mutual_information} and diagram \ref{fig:enter-label}.
Also, if $I(X; Y) = 0$, $X$ and $Y$ are independent.

Mutual information can be expressed through KL divergence \eqref{mi_kl_divergence}.

\begin{equation}
    \label{mi_kl_divergence}
    I(X;Y) = D(p_{X,Y} \parallel p_Xp_Y)
\end{equation}

As can be seen in Equation \eqref{mi_kl_divergence}, mutual information is the distance or the divergence between two probability distributions. 
If $I(X;Y) = 0$, then $p_{X,Y}$ is equal to $p_Xp_Y$.


%%% Properties of MI
\subsection{Properties of Mutual Information}
\begin{theorem} Mutual information is symmetric.
\end{theorem}
\begin{proof}
    \begin{align}
        I(X;Y) =& H(X)-H(X|Y)\\
        =& H(X)-(H(X,Y)-H(Y))\\
        =& H(X)+H(Y)-H(X,Y)\\
        =& I(Y;X)
    \end{align}
\end{proof}

\begin{theorem} Mutual information is non-negative.
\end{theorem}
\begin{proof}
    \begin{align}
        H(X)-H(X|Y) =& \Expec{\log \frac{1}{p_X(X)}}-\Expec{\log \frac{1}{p_{X|Y}(X|Y)}}\\
        =& \Expec{\log \frac{p_{X|Y}(X|Y)}{p_X(X)}}\\
        =& \Expec{\log \frac{p_{X,Y}(X,Y)}{p_X(X)p_Y(Y)}}\\
        =& \sum_{x,y} {p_{X,Y}(x,y)}\log \frac{p_{X,Y}(x,y)}{p_X(x)p_Y(y)}\\
        =& D(p_{X,Y}||p_Xp_Y) \geq 0\ (\because \textnormal{KL divergence is non-negative.})
    \end{align}
\end{proof}
Note that $p_Xp_Y$ is the distribution in $(X, Y)$ where $X$ and $Y$ have the same marginal $p_X$ and $p_Y$, but they are independent.

You can also interpret inequality $H(X) \geq H(X|Y)$ as ``conditioning always decreases (or maintains) the amount of uncertainty.



\begin{theorem}[Data processing inequality II]
For any function $f:\cX \rightarrow \fR$, we have $I(X;Y) \geq I(f(X);Y)$.
\end{theorem}
\begin{proof}
\begin{align}
I(X;Y) &= H(Y) - H(Y|X) \\
&= H(Y) - H(Y|X,f(X)) \\
&\geq H(Y) - H(Y|f(X)) \\
&= I(f(X); Y) \label{eq:dataprocessing f}
\end{align}
\end{proof}

We can also generalize to $X-Y-Z$ Markov chains (or Markov triplet), which are not special situations like the above.
Let $X-Y-Z$ form a Markov chain (or $X$ and $Z$ are conditionally independent given $Y$), we can say the following.

\begin{align}
    X-Y-Z \leftrightarrow& X \text{ and } Z \text{ are independent given } Y\\\
    \leftrightarrow & X \text{ is useless to estimate } Z \text{ When } Y \text{ is known}\\
    \leftrightarrow & p_{Z|X,Y}(z|x,y) = p_{Z|Y}(z|y) \quad\mbox{for all $x, y, z$}
\end{align}

\begin{theorem}[Data processing inequality III]
    If $X-Y-Z$ forms a Markov chain, then $I(Z;X) \geq I(Z;Y)$.
\end{theorem}
\begin{proof}
\begin{align}
I(Y;Z) &= H(Z) - H(Z|Y) \\
      &= H(Z) - H(Z|X,Y) \\
      &\geq H(Z) - H(Z|X) \\
      &= I(X;Z)
\end{align}
\end{proof}

\subsection{Conditional Mutual Information}
We can define the conditional mutual information $I(X;Z|Y)$, which is the mutual information between $X$ and $Z$, when $Y$ is given.
\begin{align}
    I(X;Z|Y) =& H(X|Y) - H(X|Y, Z)\\
    =& H(Z|Y) - H(Z|Y, X).
\end{align}

What conditions make $I(X; Z|Y )$ become 0? 
Similarly to the previous explanation of mutual information, when $X$ and $Z$ are independent given $Y$, $I(X;Z|Y) = 0$.
This is the same as the definition of a Markov chain.

\begin{remark}
There is no general relationship between $I(X;Z|Y)$ and $I(X;Z)$.
This can be examined in the following case.

\begin{equation}
    X = 0 \text{ or } 1, \text{Each probability is 1/2}
\end{equation}
\begin{equation}
    Z = 0 \text{ or } 1, \text{Each probability is 1/2}
\end{equation}
\begin{equation}
    Y = X \oplus Z
\end{equation}

In this case, $I(X;Z|Y)$ is 1 and $I(X;Z)$ is 0, so $I(X;Z|Y)$ is greater than $I(X;Z)$.
In the opposite case, we can look at the case where $I(X;Z|Y)$ is 0 and $I(X;Z)$ is greater than 0 in a markov chain.
\end{remark}

We can also define mutual information when we have multiple random variables.
For example, $I(X,Y;Z)$ means mutual information between $(X,Y)$ and $Z$.
This can be expressed as follows according to the chain rule: $I(X;Z) + I(Y;Z|X)$

$I(X^n;Y)$ can be expressed as \eqref{xn} below.
\begin{equation}
\begin{aligned}
    \label{xn}
    I(X^n;Y) &= I(X_1;Y) +I(X_2;Y|X_1)+I(X_3;Y|X_1,X_2) +\dots\\ 
    &=\sum_{i=1}^{n} I(X_i;Y|X^{i-1})
\end{aligned}
\end{equation}

\subsection{Probability Density Function}
If $X$ is continuous random variable, we can define a probability density, instead of probability mass.
Define a cumulative distribution function $F_X(x)$ where
\begin{align}
    F_X(x) = \Pr{X\leq x}.
\end{align}
If $F_X$ is differentiable, the probability density function is simply
\begin{align}
    f_X(x) = \frac{\partial}{\partial x} F_X(x).
\end{align}

Unlike probability mass function, the pdf requires additional attention during change of variables.
If $Y= aX$, we have
\begin{align}
    f_Y(y) = \frac{1}{|a|} f_X(y/a).
\end{align}
This is because the probiliby should be conserved, i.e., $|f_X(x)dx| = |f_Y(y) dy|$ where $|dx/dy| = 1/ |a|$.
More generally, let $Y = g(X)$ for a bijective function, we have
\begin{align}
    f_Y(y) = f_X(g^{-1}(y)) |\frac{d}{dy} g^{-1}(y)|.
\end{align}

\subsection{Differential Entropy}
In discrete setting, entropy becomes maximum
if the probability distribution is uniform. 
If the number of events becomes larger, then the entropy increases. 
However, there are some obstacles to be tackled for the extension from 
discrete setting to continuous settings. 
This is why we first define the mutual information for continuous random variables.

Before defining mutual information, we first extend our definition of KL divergence in continuous setting.
\begin{align}
    D(f\parallel g)=&\Esub{f}{\log{\frac{f(X)}{g(X)}}}
\end{align}
Similar to discrete case, KL-divergence is non-negative.
\begin{align}
    D(f\parallel g)=&\Esub{f}{\log{\frac{f(X)}{g(X)}}}\\
    =& \int f(x)\cdot \log{\frac{f(x)}{g(x)}}\, dx \\
    =& -\int f(x)\cdot \log{\frac{g(x)}{f(x)}}\, dx\\
    \ge & -\log{(\int \cancel{f(x)}\cdot \frac{g(x)}{\cancel{f(x)}}\, dx)}\\
    =0
\end{align}
Minus log is convex function, we can apply Jensen’s inequality .
Equality occurs in the manner of Jensen’s rule. 


To extend the natural definition of mutual information to continuous scenario, 
we see the mutual information in terms of KL-divergence. 
The continuous version of mutual information is as follows. 
\begin{equation}
\begin{split}
I(X;Y)&=\mathbb{E}\left[ \log{f_{X,Y}(X,Y) \over f_X(X)f_Y(Y)} \right] \\
      &=D(f_{X,Y}||f_Xf_Y)
\end{split}
\end{equation}

\begin{theorem} [Non-negativity of Mutual Information]
\begin{align}
    I(X;Y)\ge0
\end{align}  
\end{theorem}


\begin{proof} 
According to the definition of mutual information, it can be expressed as the KL-Divergence between $f_{X,Y}$ and $f_X\times f_Y$.
    \begin{align}
        I(X;Y) 
        =& D(f_{X,Y}\parallel f_X\times f_Y) \\
        \geq& 0.
    \end{align}
\end{proof}
This implies $h(X) \geq h(X|Y)$ where conditioning also decreases (non-increases) the differential entropy.



If we discretize the continuous random variables $X$ and $Y$ to $X^{\Delta}$ and 
$Y^{\Delta}$, we can write the definition of mutual information as follows, 
where $P(X^{\Delta})=P(i\cdot\Delta \leq  X \leq (i+1)\cdot \Delta)=\Delta \cdot f_X(X)$. 
\begin{equation}
\begin{split}
I(X^{\Delta};Y^{\Delta}) &= \mathbb{E} \left[ \log{P_{X^{\Delta}, Y^{\Delta}}(X^{\Delta}, Y^{\Delta}) \over P_{X^{\Delta}}(X^{\Delta})P_{Y^{\Delta}}(Y^{\Delta})} \right] \\
       &= \mathbb{E} \left[ \log{ \Delta^2 \cdot f_{X,Y}(X,Y) \over  \Delta \cdot f_{X}(X)\Delta\cdot f_{Y}(Y)} \right] \\    
       &= \mathbb{E} \left[ \log{ f_{X,Y}(X,Y) \over f_X(X) \cdot f_Y(Y)} \right]
\end{split}
\end{equation}
Like this, we can naturally derive continuous version of mutual information from discrete 
version of mutual information.

Differential entropy($h$) is the continuous correspondence of entropy for discrete 
random variable.
We use the same strategy to derive a continuous version of entropy and see the obstacles
to be tackled based on this derivation. 
\begin{equation}
\begin{split}
    H(X^{\Delta}) &= \mathbb{E}\left[ \log{1 \over P_{X^{\Delta}}(X^{\Delta})} \right] \\
                  &= \mathbb{E}\left[ \log{1 \over \Delta \cdot f_X(X)} \right] \\
                  &= \mathbb{E}\left[ \log{1 \over f_X(X)} \right] - \log \Delta \\
                  &= h(X) - \log \Delta
\end{split}
\end{equation}
where
\begin{align}
    h(X) = \int f_X(x) \log \frac{1}{f_X(x)}\, dx
\end{align}
is a differential entropy.

The differential entropy $h(X)$ is not equal to discretized version of entropy 
($H(X^{\Delta})$) with difference of $\log \Delta$.
$H(X^{\Delta})$ becomes bigger and bigger when $\Delta$ becomes smaller. 
This is intuitive since smaller $\Delta$ leads to more possibilities, 
which makes the entropy bigger. 

Unlike discrete entropy($H$) has label invariance, differential entropy does not 
assure this kind of invaraince. 
The definition of label invaraince is as follows. For the one-to-one map $f$, 
$H(X)=H(f(X))$.
For example, you have discrete random variable $X_1 \in \{1,2,3\}$ and $P(X_1=1)=0.4$, $P(X_1=2)=0.5$ and $P(X_1=3)=0.1$.
Additionally, you have discrete random variable $X_2=2X_1 \in \{2, 4, 6\}$ and $P(X_2=2)=0.4$, $P(X_2=4)=0.5$ and $P(X_2=6)=0.1$.
Since there is no diffrence of distributions between $X_1$ and $X_2$, 
the entropy $H(X_1)$ and $H(X_2)$ are same.  
However, this does not hold for continuous scenario. 
For example, $U \sim \mbox{Unif}(0,1)$ and $V=2U \sim \mbox{Unif}(0,2)$.
Then, $h(U)=\log(1-0)=\log 1=0$ and $h(V)= \log(2-0) = \log2 = 1$.

Also, note that the differential entropy can be negative.
For example, if $U\sim \mbox{Unif}(0, 1/2)$, then $h(U) = -\log 2$.
This is because differential entropy is ``normalized'' version of entropy with $\log \Delta$ term.

\subsection{Properties of Differential Entropy}
In discrete case 1 to 1 function f(x) discrete entropy are same. But differential entropy does not. 

    \begin{align}
        h(X) \neq h(aX) = h(X)+ \log{|a|}
    \end{align}

However, mutual information are same. You can interpret the mutual information in the same way. Amount of information $X$ tells about the $Y$ or the amount of information $aX$ tells about the $Y$. They should be the same and that's also the case in the continuous case.

\begin{theorem}  [Property of Mutual Entropy 1]
\begin{align}
    I(aX;Y)=I(X;Y)
\end{align}  
\end{theorem}
      
\begin{proof}
    \begin{align}
        I(aX;Y)
        =&h(aX)-h(aX | Y)\\
        =&(h(X) + \log{|a|}) \\
        -&(h(X|Y) + \log(|a|)) \\
        =& I(X;Y)
    \end{align}
\end{proof}

\begin{remark}
    In the above proof, we need to define conditional differential entropy.
    The definition is consistent with conditional entropy:
    \begin{align*}
        h(X|Y)=&\Expec{\log{\frac{1}{f_{X|Y}(X|Y)}}}\\
        =&\int f_{X, Y}(x, y) \log{\frac{1}{f_{X|Y}(x|y)}}\,dx dy.
    \end{align*}
    In particular, we have
    \begin{align*}
        h(aX|Y)=&\Expec{\log{\frac{1}{f_{X|Y}(aX|Y)}}}\\
         =&  \Expec{\log{\frac{1}{f(X|Y)}}}+\log{|a|}
    \end{align*}
\end{remark}

\subsection{Joint Differential Entropy}

\begin{theorem}  [Chain Rule of Differential Entropy]     
\begin{align}
h(X_1, X_2) = h(X_1)+h(X_2|X_1)
\end{align}
\end{theorem}

\begin{proof} 
Let's denote $X_1,X_2,X_3,...X_n$ as $X$. Then we may need to define a differential entropy entropy for joint distributions. 

For example, if $X_1$ and $X_2$ are continuous random variables,
    \begin{align}
        h(X_1|X_2)=&\Expec{\log{\frac{1}{f_{X_1,X_2}(X_1,X_2)}}}\\
        =& \Expec{\log{\frac{1}{f_{X_1}(X_1)}}}+\Expec{\log{\frac{1}{f_{X_2|X_1}(X_2|X_1)}}}\\
        =& h(X_1)+h(X_2|X_1)
    \end{align}
   This is using the same property of the pmf.
   Joint is a product between the marginal times the conditional which is also the case on the pmf.
\end{proof}

\begin{theorem} 
$X$ and $Y$ are independent if and only if
\begin{align}
    I(X;Y)=0
\end{align}  
\end{theorem}

\begin{theorem} [Data processing Inequality]
If $X-Y-Z$ form a Markov chain, then
\begin{align}
    I(X;Y) \ge I(Z;X)
\end{align}  
\end{theorem}

\begin{proof}
    Since conditioning decreases (non-increases) the entropy.
    \begin{align*}
        I(X;Y)=&h(Z)-h(Z|Y)\\
        =& h(Z)-h(Z|Y,Z)\\
        \ge& h(Z)-h(Z|X)\\
        =& I(X;Z)
    \end{align*}
     where the first equality is from Markov property $f(Z|Y)=f(Z|Y,X)$.
\end{proof}

\subsection{Maximum Differential Entropy} \label{sec:intro-cross}
For $X\in \cX=\{1,2,3,\ldots,K\}$, we showed $H(X)\le \log_{2}{K}$ where the equality holds for uniform distribution.
In other words, the maximum entropy is achieved by an uniform distribution in discrete case.
Then, what about $\max h(X)$

\begin{exercise}
    If $X\in[a, b]$ with probability 1, what is the maximum differential entropy?
\end{exercise}

If we do not have bounds, then it can be infinite.
So we need a restriction for the random variable $X$.
A common constraint is $\Expec{X^2} \le P$.
Under this second moment constraint, what is the maximum of differential entropy? 

\begin{theorem}
    The Gaussian distribution achieves maximum differential entropy subject to the second-moment constraint.
\end{theorem}
\begin{proof}
Let $X$ be a random variable with pdf $f_X$ where $\Esub{f}{X^2} = P$,
while $X'$ be a Gaussian random variable $\cN(0, P)$ where pdf is $g(x) = \frac{1}{2\pi P} \exp(-x^2/(2P))$.
Then consider the KL divergence between $f$ and $g$.

    \begin{align}
        D(f\parallel g)
        =& \mathify{\mathbb{E}_f\left[{\log{\frac{f_X(X)}{g(X)}}}\right]}\\
        =& \Esub{f}{\log \frac{1}{g(X)}} -  \Esub{f}{\log \frac{1}{f_X(X)}}\\
        =&\Esub{f}{\log \frac{1}{g(X)}} -h(f_X)
    \end{align}
    where $h(f) = h(X)$ with $X\sim f_X$.
    The first term is simply
    \begin{align*}
    \Esub{f}{\log \frac{1}{g(X)}} 
        =& \log{\sqrt{2\pi P}} +\mathify{\mathbb{E}_f\left[{\frac{X^2}{2P}}\right]}\\
        =& \log{\sqrt{2\pi P}} +\mathify{\mathbb{E}_g\left[{\frac{X^2}{2P}}\right]}\\
        =&\mathify{\mathbb{E}_g\left[{\log{\frac{1}{g(X)}}}\right]}\\
        =&h(g).
    \end{align*}
Therefore, we see that
\begin{align*}
    D(f\parallel g) = h(g)-h(f_X) \ge 0
\end{align*}
    by the non-negativity of divergence.
    Therefore,
\begin{align*}
     h(g) \ge h(f)
\end{align*}
\end{proof}
Thus, under the second moment constraint, Gaussian distribution is ``maximally'' random.

\section{Random Process}
% \url{https://brilliant.org/wiki/stationary-distributions/}
We define random process $\bX = \{X_n\}_{n=1}^\infty$ as a sequence of random variables. 
The most simple random process is an independent and identically distributed (i.i.d.)~process.
For i.i.d.~process, we have
\begin{align}

\end{align}

In practice, for example, in text, symbols are highly correlated (e.g., one can easily deduce the missing letter is \texttt{m}). 
Therefore, such sequence does not follow an i.i.d.~distribution.

Markov means there is first order correlation ({\tt 1st order markov}).

X_0=0, X_n = \begin{cases}

\end{example}

\subsection{$1^\mn{st}$ Order Markov Process}

\begin{definition}
Consider a random process $\bX$. 
If the probability of the $i^\mn{th}$ step $P(X_i|X_\mn{i-1},\cdots,X_1)=P(X_i|X_\mn{i-1})$,
the sequence is said to follow a $1^\mn{st}$ order Markov process.
\end{definition}

\begin{remark}
Consider the joint probability distribution of all possible tuples in the sequence denoted as $P_{X^n}(x^n)=P_{X_1,\cdots,X_n}(x_1, \dots, x_n)$.
Then, the first order Markov process satisfies
\begin{align}
     P_{X^n}(x^n) =& \prod_{i=1}^{n} P_{X_i|X^\mn{i-1}}(x_i|x^{i-1})\\
    =& \prod_{i=1}^{n} P_{X_i|X_\mn{i-1}}(x_i|x_{i-1})
\end{align}
\end{remark}

Consider a random process $\bX$ where $X_i\in\cX = \{1, \dots, n\}$ for all $i$.
Suppose that the transition probability $P_{X_i|X_{i-1}}(x_i|x_{i-1})$ remains the same for all $i$.
Then, we can define a $n\times n$ transition matrix $P$ where the $(u, v)$-th element is $P_{u, v} = \Pr{X_i=u|X_{i-1}=v}$.
All elements in $P$ are nonnegative and the sum of elements of each column in the transition matrix is 1.

Let $\pi_t$ is an $n$-dimensional vector where
\begin{align}
    \pi_i = \begin{bmatrix}
        \Pr{X_i=1}\\
        \Pr{X_i=2}\\
        \vdots \\
        \Pr{X_i=n}\\
    \end{bmatrix}
\end{align}
Then, we have
\begin{align}
    \pi_i = P \times \pi_{i-1}
\end{align}
for all $i\geq 1$.

\begin{example}
    Consider a binary random process where $p_{X_i|X_{i-1}}(0|1)=p_{X_i|X_{i-1}}(1|0) = \alpha<1/2$ and $p_{X_i|X_{i-1}}(1|0)=p_{X_i|X_{i-1}}(0|1) = 1-\alpha$.
    Then we can define a transition matrix
    \begin{align}
        P = \begin{pmatrix}
            1-\alpha & \alpha\\
            \alpha & 1-\alpha
        \end{pmatrix}.
    \end{align}
    Suppose we have $\pi_0 = [1, 0]$, namely $X_0=0$ with probability 1.
    Then, $\pi_1 = [1-\alpha, \alpha]$.
\end{example}

\subsection{$k^\mn{th}$ Order Markov Process}
\begin{definition}
Consider a random process $\bX$.
If $P_{X_i|X^\mn{i-1}} (x_i|x^{i-1})=P_{X_i|X_\mn{i-k}^\mn{i-1}} (x_i|x_{i-k}^{i-1})$, 
the sequence is said to follow the $k^\mn{th}$ order Markov process.
\end{definition}

\begin{remark}
If the sequence follows the $k^\mn{th}$ order Markov, then $P_{X^n}(x^n)=\prod_{i=1}^{n} P_{X_i|X_\mn{i-k}^\mn{i-1}}(x_i|x^{i-1}_{i-k})$.
\end{remark}

\subsection{Stationary Distribution}
\begin{figure}[htpb]
\begin{center}
\includegraphics[scale=0.5]{images/scribe12-distribution.png}
\caption{The assumption we've learned ordered by proximity to practical setting. The i.i.d assumption considers no correlation between the random variables in a sequence, assuming complete independence, while the practical settings have higher correlation among the sequence's random variables. A stationary distribution is a closer approximation to practical than a $k^\mn{th}$ order Markov process.}
\end{center}
\end{figure}
\begin{definition}
Consider the random process $X_1,X_2,\cdots,X_n$. The process is said to be stationary 
if $X_i^n$ is distributionally equivalent with $X_\mn{i+1}^\mn{i+n}$ for all $i$ and $n$.
\end{definition}

Note that the $P_{X_i^n}$ denotes the probability distribution of $n-$tuples starting from the $i^\mn{th}$ step,
and $P_{X_\mn{i+1}^\mn{i+n}}$ denotes the probability distribution of $n-$tuples starting from the $(i+1)^\mn{th}$ step.
That is, if we take any $n-$tuples starting from $i^\mn{th}$ step and shift them to start from $(i+1)^\mn{th}$ step,
they follow the same distribution if the process is stationary. Thus, if the process is stationary,
the random variable of the process does not depends on time. 

\begin{example}
Consider the random walk process :
\begin{equation}
X_0=0, \quad X_n=X_\mn{n-1}\pm1
\end{equation}
then the possible value of $X_1$ and $X_2$ is:
\begin{equation}
        X_1 = \begin{cases} 0\\ -1, \end{cases}
    X_2 = \begin{cases} 2\\ 0\\ -2 \end{cases}
\end{equation}
Therefore we can expect the $X_n$ is ``more random" than $X_0$. Thus the process is not stationary.
\end{example}

\begin{remark}
    Not all Markov process is stationary.
    Also, stationary process can have infinite dependencies.
\end{remark}

\subsection{Stationary Markov Process}

\begin{example}\label{ex:markovone}
Consider the $1^\mn{st}$ order Markov process with the initial distribution $P(A)=P(B)=P(C)=\frac{1}{3}$.
\begin{figure}[htpb]
\begin{center}
\includegraphics[scale=0.3]{images/scribe12-markovex.png}
\caption{$1^\mn{st}$ order Markov transition.}
\label{fig:markovtrans}
\end{center}
\end{figure}
\end{example}

Then we can have
\begin{equation}
\begin{split}
    & P(X_1=A)=P(X_1=B)=P(X_1=C)=\frac{1}{3} \\
    & P(X_2=A)=P(X_1=A)\cdot P(X_2=A|X_1=A) \\
    & +P(X_1=C)\cdot P(X_2=A|X_1=C)=\frac{1}{3}
\end{split}
\end{equation}
When the same process is applied to $P(X_2=B)$ and $P(X_2=C)$, the probability consistently remains at $\frac{1}{3}$, and this holds true when extended to $X_3, X_4, \cdots$ where all probabilities are $\frac{1}{3}$ as well. Thus, the probabilities at $X_i$ always follow the initial probability of $\frac{1}{3}$, indicating that the process is stationary: if we take any $n-$tuples in the process starting from $i^\mn{th}$ step and shift to $(i+1)^\mn{th}$ step, the distribution will be always same.

\begin{example}
Consider the same process described in Example \ref{ex:markovone}, with the initial distribution $P(X_1=A)=1$ and $P(X_2=A)=P(X_2=B)=\frac{1}{2}$. Given this condition, the process is no longer stationary. That is, the Markov process itself does not guarantee stationary; however, under certain assumptions, Markov process can have stationary initial distributions, as in the case of Example \ref{ex:markovone}.
\end{example}

\begin{theorem}
With some reasonable assumptions (finite state, irreducible, aperiodic), Markov process can have a stationary initial distribution.
\end{theorem}

\begin{example}
    Consider a binary random process where $p_{X_i|X_{i-1}}(0|1)=p_{X_i|X_{i-1}}(1|0) = \alpha<1/2$ and $p_{X_i|X_{i-1}}(1|0)=p_{X_i|X_{i-1}}(0|1) = 1-\alpha$.
    Then we can define a transition matrix
    \begin{align}
        P = \begin{pmatrix}
            1-\alpha & \alpha\\
            \alpha & 1-\alpha
        \end{pmatrix}.
    \end{align}
    Suppose we have $\pi_0 = [1/2, 1/2]$, then it is not hard to show that $\pi_t = [1/2, 1/2]$ for all $t$.
\end{example}

The stationary distribution $\pi^\star$ should satisfy
\begin{align}
    \pi^\star = P \times \pi^\star.
\end{align}
Thus, the stationary distribution is an eigenvector of $P$ corresponds to eigenvalue 1.
Recall that the eigenvalue $\lambda$ of matrix $P$ is a solution of $\det(P-\lambda I)=0$.

\begin{exercise}
    Show that the uniform distribution is a stationary distribution if $P$ is symmetric.
\end{exercise}

We let $\pi_\infty$ be the limiting distribution (if exsits), i.e.,
\begin{align}
    \pi_\infty = \lim_{t\rightarrow \infty} \pi_t.
\end{align}
Note that the limiting distribution may depend on initial distribution $\pi_0$.
\begin{theorem}
    The limiting distribution should be a stationary distribution.
\end{theorem}
The proof is simply taking a limit on both side of $\pi_{t+1} = P \pi_t$.

\begin{example}[Discrete Diffusion]
    Given $X_0\sim p_0$ be the random variable with data distribution $p_0$.
    Consider a transition matrix 
    \begin{align}
        P = \begin{bmatrix}
        1-\epsilon & \epsilon /(n-1)& \dots &\epsilon /(n-1)\\
        \epsilon/(n-1) & 1-\epsilon & \dots &\epsilon /(n-1)\\
        &\vdots&&\\
        \epsilon/(n-1) & \epsilon/(n-1)& \dots &1-\epsilon \\
       \end{bmatrix} 
    \end{align}
    Let $\bX$ be a 1-st order Markov process where $\pi_{t+1} = P \pi_t$.
    The limiting distribution is a uniform distribution $\pi^\star = [1/n, \dots, 1/n]$.
    For sufficiently large $N$, we can say that $\pi_N \sim \pi^\star$.
    On the other hand, if we can train a neural network that can recover (denoise) $X_t$ from $X_{t+1}$ for some $t$, i.e.,
    \begin{align}
        X_t \approx f_\theta(X_{t+1}, t).
    \end{align}
    Then, the generation process is the following.
    Sample $\tilde{X}_N$ according to the uniform distribution, and apply $f_{\theta}(\cdot, t)$ recursively to obtain $\tilde{X}_0$.
    Then, we can expect that $\tilde{X}_0$ behaves similarly to $X_0\sim p_0$.
    This is a key idea of generative diffusion process.
\end{example}

\section{Expectation Review}
\begin{theorem}[Markov Inequality]
If $X$ is a non-negative random variable, then for any $\alpha > 0$,
\begin{align}
    \Pr{X\geq \alpha \Expec{X}} \leq \frac{1}{\alpha}.
\end{align}
\end{theorem}

\begin{theorem}[Chebyshev Inequality]
Let $X$ be a variable with mean $\mu$ and variance $\sigma^2$.
For any $\alpha > 0$,
\begin{align}
    \Pr{|X- \Expec{X} | \geq \alpha\sigma} \leq \frac{1}{\alpha^2}.
\end{align}
\end{theorem}

\begin{definition} [Moment generating function]
For random variable $X$, the moment generating function is
\begin{align}
    M_X(t) = \Expec{e^{tX}}.
\end{align}
\end{definition}

The reason why it is called moment generating function is the following.
\begin{align}
    \frac{\partial}{\partial t}M_X(t) =& \Expec{\frac{\partial}{\partial t} e^{tX}}\\
    =& \Expec{X e^{tX}}
\end{align}
which implies $M_X'(0) = \Expec{X}$.
Also, 
\begin{align}
    \frac{\partial^2}{\partial t^2}M_X(t) =& \Expec{\frac{\partial^2}{\partial t^2} e^{tX}}\\
    =& \Expec{X^2 e^{tX}}
\end{align}
which implies $M_X''(0) = \Expec{X^2}$.

\begin{exercise}
    Show that the moment generating function of Gaussian random variable $X\sim\cN(\mu, \sigma^2)$ is given by
    \begin{align}
        M_X(t) = \exp\left(\frac{t^2\sigma^2}{2} + t\mu  \right).
    \end{align}
\end{exercise}

\begin{theorem}
    For two random variables $X$ and $Y$, if $M_X(t) = M_Y(t)$ for all $t$, then $X$ and $Y$ have the same distribution.
    More precisely, they have the same cumulative distribution
    \begin{align}
        F_X(x) = F_Y(x)
    \end{align}
    for all $x$.
\end{theorem}
The proof is out of the scope of this lecture.

For independent random variable $X$ and $Y$, let $Z= X+Y$, then
\begin{align}
    M_Z(t) = \Expec{e^{tZ}} 
    = \Expec{e^{t(X+Y)}} 
    = \Expec{e^{tX}} \Expec{e^{tY}} 
    = M_X(t) M_Y(t).
\end{align}

\begin{definition}
    Let $X$ be a Gaussian distribution with mean $\mu$ and variance $\sigma^2$, then we say $X\sim\cN(\mu, \sigma^2)$ and the corresponding probability density function (pdf)
    \begin{align}
        f_X(x) = \frac{1}{\sqrt{2\pi \sigma^2}} \exp\left(-\frac{(x-\mu)^2}{2\sigma^2}\right).
    \end{align}
\end{definition}

\begin{exercise}
    Show that the sum of independent Gaussian $X\sim\cN(\mu_1, \sigma_1^2)$
\end{exercise}

\begin{remark}
    We can prove (non-rigorously) Central Limit Theorem (CLT) using MGF.
    Let $X_1,\dots $ be i.i.d.~random process with mean $\mu$ and variance $\sigma^2$.
    Then, the CLT tells you
    \begin{align}
        \sum_{i=1}^n \frac{X_i-\mu}{\sqrt{n}\sigma} \stackrel{(d)}{\longrightarrow } \cN(0, 1).
    \end{align}
    Consider $Y = (X-\mu)/\sigma$.
    Then, we have $\Expec{Y} = 0$ and $\Expec{Y^2} = \Var{Y} = 1$, which implies $M_Y'(0) = 0$ and $M_Y''(0) = 1$.
    The Taylor approximation tells us $M(Y)(t) \approx M_Y(0) + M_Y'(0)t + \frac{M_Y''(0)}{2}t^2 = 1 + t^2/2$.
    Then, let $Z_n = \frac{1}{\sqrt{n}}\sum_{i=1}^n Y_i$, where the moment generating function of $Z_n$ is
    \begin{align}
        M_{Z_n}(t) =& \prod_{i=1}^n \Expec{\exp(tY_i/\sqrt{n})}\\
        =& M_Y(t/\sqrt{n})^n\\
        \approx& (1 + t^2/2n)^n\\
        \rightarrow& e^{t^2/2}.
    \end{align}
\end{remark}

\begin{theorem}[Chernoff Inequality]
Let $X$ be a variable with moment-generating function $M(t) = \Expec{e^{tX}}$.
For any $\alpha > 0$,
\begin{align}
    \Pr{X\geq \alpha} \leq M(t) e^{-t\alpha}
\end{align}
for all $t>0$.
\end{theorem}

\begin{remark}
    Let $X_1, \dots, X_n$ be i.i.d.~process with mean $\mu$ and variance $\sigma^2$.
    The weak law of large numbers (LLN) tells you that
    \begin{align}
        \frac{1}{n}\sum_{i=1}^n X_i \rightarrow \mu
    \end{align}
    where the convergence is in probability (it also converges with probability 1 in strong LLN).

    Let $Z_n = \frac{1}{n}\sum_{i=1}^n X_i$, then the $Z_n$ is a random variable with mean $0$ and variance $\sigma^2/n$.
    \begin{align}
        \Pr{|Z_n-\mu| > \epsilon} =&  \Pr{|Z_n-\mu| > \frac{\epsilon}{\sigma/\sqrt{n}} \frac{\sigma}{\sqrt{n}}} \\
        \leq& \frac{\sigma^2}{n\epsilon}
    \end{align}
    which converges to 0.
\end{remark}

\begin{definition}
    The conditional expectation of $X$ given $Y=y$ is
    \begin{align}
        \Expec{X|Y=y} = \sum_{x} x p_{X|Y}(x|y)
    \end{align}
    or
    \begin{align}
        \Expec{X|Y=y} = \int_{x} x f_{X|Y}(x|y)\, dx.
    \end{align}
\end{definition}
Conditional expectation is an expectation when the distribution is conditional $p_{X|Y}(\cdot|y)$ or $f_{X|Y}(\cdot|y)$.
Thus, when we say $\Expec{X|Y}$ it is a function $Y$, and also random (since $Y$ is random).

\begin{theorem}
    \Expec{\Expec{X|Y}} = \Expec{X}.
\end{theorem}

\section{Estimation}
Let $X$ be a ``sample'' (input), and $Y$ be the label (output).
Assume that $p_{X|Y}(x|y)$ is known.
Then, the question is to estimate $y$ given $X=x$.

\begin{example}\label{example:estimation}
Suppose we have two nationalities, A and B.
The height distribution for individuals from nationality A follows a normal distribution $\mathcal{N}(170, 10^2)$, while for nationality B, it follows $\mathcal{N}(180, 15^2)$.
In this scenario, we define $Y$ as a binary label, where $Y=0$ corresponds to nationality A and $Y=1$ corresponds to nationality B, and $X$ represents the individual's height.

We are interested in a classification task where the goal is to predict the nationality based on the height.
For example, given a height of $X=176$, how can we estimate the corresponding $Y$?
\end{example}

%%% Maximum Likelihood Estimation (MLE)
\subsection{Maximum Likelihood Estimation (MLE)}
The maximum likelihood estimator finds $Y=y$ that maximizes the likelihood function, i.e.,
\begin{align}
    \argmax_y f_{X|Y}(x|y)
\end{align}
In words, it asks ``on which $y$, the given $X=x$ is the most likely.''
In Example~\ref{example:estimation}, we pick $y=0$ if $p_{X|Y}(x|0) > p_{X|Y}(x|1)$.
In other words,
\begin{align}
    \frac{1}{\sqrt{2\pi 10^2}}\exp\left(-\frac{(x-170)^2}{2\cdot 10^2}\right)
    >\frac{1}{\sqrt{2\pi 15^2}}\exp\left(-\frac{(x-180)^2}{2\cdot 15^2}\right).
\end{align}


%%% Maximum a posteriori estimation
\subsection{Maximum A Posteriori (MAP) Estimation}
Maximum a posteriori (MAP) estimator finds $Y=y$ that maximizes the posterior probability $p_{Y|X}(y|x)$, i.e.,
\begin{align}
    \argmax_y p_{Y|X}(y|x)
\end{align}
It asks ``given $X=x$, which $y$ is the most probable.''
Note that MAP requires the knowledge of $p_Y(y)$ for all $y$.
If the $Y$ is a continuous random variable, replace the pmf with $f_{Y|X}(y|x)$.

\begin{question}
    What about $p_X(x)$?
\end{question}

If we apply Bayes' rule, we have
\begin{align}
    p_{Y|X}(y|x) = \frac{p_{X|Y}(x|y)p_Y(y) }{p_X(x)}.
\end{align}
Since $p_X(x)$ is constant (not a function of $y$, MAP rule is simply
\begin{align}
    \argmax_y p_{X|Y}(x|y)p_Y(y).
\end{align}
Thus, if $p_Y(y)$ is uniform, MAP estimation is equivalent to MLE.

In Example~\ref{example:estimation}, suppose the population from A is double, i.e., $p_Y(0) = 9/10$ and $p_Y(1)=1/10$.
Then, MAP estimator pick $y=0$ if
\begin{align}
    \frac{1}{\sqrt{2\pi 10^2}}\exp\left(-\frac{(x-170)^2}{2\cdot 10^2}\right) \times \frac{9}{10}
    >\frac{1}{\sqrt{2\pi 15^2}}\exp\left(-\frac{(x-180)^2}{2\cdot 15^2}\right) \times \frac{1}{10}.
\end{align}

\begin{theorem}
    MAP is Bayes optimal. In other words, it minimizes the probability of error.
\end{theorem}
If the goal of the estimation is to minimize the probability of error $\Pr{\hat{Y}\neq Y}$, then
\begin{align}
    \hat{Y}_{MAP}(X) =  \argmax_y p_{X|Y}(x|y)
\end{align}
is the minimum achiever.

%%% Fano's Inequality
\subsection{Fano's Inequality}
Let $Y\in\cY$ be a true label and $X$ be an input (image).
We want to estimate $\hat{Y}(X)$ as a funciton of $X$
where the we focus on hard classfication namely $\hat{Y}(X)$ is also discrete $\hat{Y}(X)\in\cY$.
Then, we are interested in the probability of error $\Pr{Y\neq \hat{Y}(X)}$.

\begin{theorem}
The probability of error is bounded by
\begin{align}
    \Pr{Y\neq \hat{Y}(X)} \geq \frac{H(Y|X)-1}{\log | \cY |}
    \end{align}
\end{theorem}
\begin{proof}
    Let $E$ be a binary random variable where $E=1$ if $\hat{Y}(X)\neq Y$ and $E=0$ if $\hat{Y}(X) = Y$.
    Then,
    \begin{align}
        H(E, Y|\hat{Y}) =& H(Y|\hat{Y}) + H(E|Y, \hat{Y})\\
        =& H(Y|\hat{Y})\\
        \geq& H(Y|X).
    \end{align}
    where the last inequality is due to data processing inequality since $\hat{Y}$ is a function of $X$.

    On the other hand,
    \begin{align}
        H(E, Y|\hat{Y}) =& H(E|\hat{Y}) + H(Y|E, \hat{Y})\\
        \leq& 1 + p_E(1) H(Y|E=1, \hat{Y}) + p_E(0) H(Y|E=0, \hat{Y})\\
        =& 1 + p_E(1) H(Y|E=1, \hat{Y})\\
        \leq& 1 + p_E(1) \log |\cY|.
    \end{align}
    Since $p_E(1) = \Pr{Y\neq \hat{Y}(X)}$, we got
    \begin{align}
        H(Y|X) \leq 1 + \Pr{Y\neq\hat{Y}(X)} \log |\cY|,
    \end{align}
    which concludes the proof. 
    \end{proof}
    The above theorem tells us that the probability of error cannot be small if $Y|X$ has too much randomness.


%%% Parameter Estimation
\subsection{Parameter Estimation}
Suppose $X_1, \dots, X_n$ is i.i.d.~process with $p_X$ where $p_X$ is parameterized by $\theta$.
We are interested in estimating $\theta$ given samples $x_1, \dots, x_n$.
Without assumption on distribution of $\theta$, we can employ Maximum Likelihood Estimator.
In other words, we find $\theta$ that maximizes $p_{X^n|\theta}(x^n|\theta)$.
Since maximizing the likelihood is equivalent to maximize the log-likelihood, we often maximize
\begin{align}
    \sum_{i=1}^n \log p_{X_i|\theta} (x_i|\theta).
\end{align}

\begin{example}
    Suppose $X_1, \dots, X_n$ is i.i.d.~Bernoulli distribution of probability $p_X(1) = p$.
    We can say that the distribution is parameterized by $\theta=p$.
    Note that the Bernoulli distribution can be represented by $p_{X}(x) = p^X(1-p)^{1-X}$ since it returns $p$ for $X=1$ and $1-p$ for $X=0$.
    Then,
    \begin{align}
        \log p_{X^n}(x^n) =& \sum_{i=1}^n \log p^{X_i}(1-p)^{1-X_i}\\
        =& \sum_{i=1}^n X_i \log p + \sum_{i=1}^n (1-X_i)\log (1-p).
    \end{align}
    If we take the derivative with respect to $p$, we have
    \begin{align}
        \frac{1}{p}\sum_{i=1}^n X_i  = \frac{1}{1-p} \sum_{i=1}^n (1-X_i)
    \end{align}
    This implies $\hat{p} = \frac{1}{n}\sum_{i=1}^n X_i$
\end{example}

\begin{example}
    Suppose $X_1, \dots, X_n$ is i.i.d.~Gaussian distribution $\cN(\mu, \sigma^2)$
    We can say that the distribution is parameterized by $\theta=(\mu, \sigma^2)$.
    Similarly, consider the log-likelihood,
    \begin{align}
        \log f_{X^n}(x^n) = \sum_{i=1}^n -\frac{(x_i-\mu)^2}{2\sigma^2} - \frac{1}{2}\log 2\pi \sigma^2.
    \end{align}
    If we take the derivative with respect to $\mu$, then 
    \begin{align}
         \sum_{i=1}^n 2\frac{x_i-\mu}{2\sigma^2} = 0
    \end{align}
    which implies $\hat{\mu} = \frac{1}{n}\sum_{i=1}^{n}X_i$.
    Also, if we take the derivative with respect to $t = \sigma^2$, then
    \begin{align}
         \sum_{i=1}^n \frac{(x_i-\mu)^2}{2t^2} = \frac{n}{2t}
    \end{align}
    Thus,
    \begin{align}
        \widehat{\sigma^2} = \frac{1}{n}\sum_{i=1}^n (x_i-\mu)^2.
    \end{align}
\end{example}



%%% Naive Bayes
\subsection{Naive Bayes}
Suppose we have $X^n = (X_1, \dots, X_n)$ is the input (image) and $Y$ be the corresponding label.
Classification is simply estimating the label $Y$ based on the input $X^n$.
We assume the probabilistic model $P_{X^n|Y}$ where $X^n$ is conditionally independent given $Y$.

\begin{example}[Spam Filter]
    Consider $X^n$ be the collection of words, where $X_i=1$ if the mail contains the $i$-th word in the dictionary (and 0 otherwise).
    Then, the Naive Bayes assumption is simply the occurence of each word given $Y=1$ (the mail is spam) is independent.
\end{example}

\begin{example}[MNIST]
    Consider $X^n$ be the binarized MNIST image, i.e., it is $7\times 7$ image where each pixel is either 0 or 1. 
    Then, the Naive Bayes assumption implies that the pixel values are independent given the label.
\end{example}

As the name suggests, Naive Bayes assumption is over-simplified probabilistic model, but it works reasonably in simple cases.
The assumption implies
\begin{align}
    \log p_{X^n|Y}(x^n|y) = \sum_{i=1}^n \log p_{X_i|Y}(x_i|y).
\end{align}
The MLE is
\begin{align}
    \hat{y}_{MLE} = \argmax_{y} \sum_{i=1}^n \log p_{X_i|Y}(x_i|y).
\end{align}
In Spam filter example, each $p_{X_i|Y}(x_i|y)$ are Bernoulli distribution with parameter $p_{i|y}$.
If we have samples $\{(\bx^{(1)}, y^{(1)}), (\bx^{(2)}, y^{(2)}), \dots, (\bx^{(N)}, y^{(N)})\}$ where $\bx^{(i)}$ is the $i$-th mail and $y^{(i)}$ is the corresponding label,
then we can estimate each parameter by
\begin{align}
    p_{i|y=0} =& \frac{\sum_{j=1}^N \bone\{\bx^{(j)}_i=1 \mbox{ and }y^{(j)}=0\}}{\sum_{j=1}^N \bone\{y^{(j)}=0\}}\\
    p_{i|y=1} =& \frac{\sum_{j=1}^N \bone\{\bx^{(j)}_i=1 \mbox{ and }y^{(j)}=1\}}{\sum_{j=1}^N \bone\{y^{(j)}=1\}}
\end{align}
Finally, we further apply MLE to estimate the label of input $x^n$.

Moreover, since $Y$ is also a Bernoulli variable, we can approximate $p_y = \Pr{Y=1}$ using MLE.
\begin{align}
    p_{y} =& \frac{1}{N}\sum_{j=1}^N \bone\{y^{(j)}=0\}.
\end{align}
Then, we can apply MAP to estimate the label of input $x^n$.
\begin{align}
    \hat{y}_{MAP} = \argmax_{y} p_{Y|X^n}(y|x^n).
\end{align}
where
\begin{align}
     p_{Y|X^n}(1|x^n) =& \frac{p_{X^n|Y}(x^n|1)p_Y(1)}{p_{X^n|Y}(x^n|0)p_Y(0) + p_{X^n|Y}(x^n|1)p_Y(1) }\\
     =& \frac{\prod_{i=1}^np_{X_i|Y}(x_i|y) p_Y(1)}{\prod_{i=1}^np_{X_i|Y}(x_i|0)p_Y(0) + \prod_{i=1}^np_{X_i|Y}(x_i|1)p_Y(1) }
\end{align}
where the probability $p_{X_i|Y}(x_i|0)$ and $p_{X_i|Y}(x_i|0)$ is estimated via MLE.


However, in the MLE above for $p_{i|y}$, some rare word $x_i$ may never appear in the sample,
resulting in $p_{i|y=0}=p_{i|y=1}=0$.
To avoid such situation, we can use Laplace smoothing, which replace the MLE with 
\begin{align}
    p_{i|y=0} =& \frac{1+\sum_{j=1}^N \bone\{\bx^{(j)}_i=1 \mbox{ and }y^{(j)}=0\}}{2+\sum_{j=1}^N \bone\{y^{(j)}=0\}}\\
    p_{i|y=1} =& \frac{1+\sum_{j=1}^N \bone\{\bx^{(j)}_i=1 \mbox{ and }y^{(j)}=1\}}{2+\sum_{j=1}^N \bone\{y^{(j)}=1\}}
\end{align}
You can think of it as you count the occurence of each word starting from 1 (instead of 0).

\subsection{Bias-Variance}
Given a sample \( X_1, \dots, X_n \), consider an estimator 
\[
\hat{\theta}_n \equiv \hat{\theta}(X_1, \dots, X_n)
\]
of a real-valued parameter \( \theta \in \Omega \subset \mathbb{R} \).

The distribution of any estimator of course depends on the true parameter vector \( \theta \), i.e., more precisely,
\[
\hat{\theta}_n \equiv \hat{\theta}(X_1, \dots, X_n; \theta).
\]

The bias of an estimator is defined by
\[
\text{Bias}(\hat{\theta}_n) = \mathbb{E}(\hat{\theta}_n) - \theta.
\]
An estimator is called \emph{unbiased} if \( \mathbb{E}(\hat{\theta}_n) = \theta \), and hence 
\[
\text{Bias}(\hat{\theta}_n) = 0
\]
for all possible \( \theta \).

The variance of an estimator is given by
\[
\Var{\hat{\theta}_n} = \mathbb{E}\left( (\hat{\theta}_n - \mathbb{E}(\hat{\theta}_n))^2 \right).
\]

Performance of an estimator is most frequently evaluated with respect to the mean squared error loss (also called \( L_2 \) loss):
\[
(\hat{\theta}_n - \theta)^2.
\]
The corresponding risk is the Mean Squared Error (MSE):
\[
\mathbb{E}\left( (\hat{\theta}_n - \theta)^2 \right) = \text{Bias}(\hat{\theta}_n)^2 + \Var{\hat{\theta}_n}.
\]
For an unbiased estimator, the mean squared error is obviously equal to the variance of the estimator.

\begin{example}
 Assume an i.i.d. sample \( X_1, \dots, X_n \) with mean \( \mu = \mathbb{E}(X_i) \) and variance \( \sigma^2 = \Var{X_i} < \infty \).
The sample mean \( \hat{X} \) is an unbiased estimator of the true mean \( \mu \), since the equation
\[
\mathbb{E}(\hat{X}) = \mu
\]
holds for any possible value of the true mean \( \mu \).

The variance of the estimator \( \hat{X} \) is given by
\[
\Var{\hat{X}} = \frac{\sigma^2}{n}.
\]
The mean squared error of the estimator \( \hat{X} \) is given by
\[
\mathbb{E}\left( (\hat{X} - \mu)^2 \right) = \Var{\hat{X}} = \frac{\sigma^2}{n}.
\]
\end{example}


%%% Minimum Mean Squared Error Estimation}
\subsection{Minimum Mean Squared Error Estimation}
Let $X$ be the source and $Y$ be the noisy measurement where we want to estimate $X$ given $Y$.
Suppose we want to find the Minimum Mean Squared Estimation $\hat{X}(Y)$ as a funciton of $Y$ that minimizes
\begin{align}
    \Expec{(X-\hat{X}(Y))^2}.
\end{align}

\begin{theorem}
    Conditioanl expectation is Bayes' optimal.
    In other words, $\hat{X}(Y) = \Expec{X|Y}$ minimizes the MSE.
\end{theorem}
\begin{proof}
\begin{align}
    \Expec{(X-\hat{X}(Y))^2}=& \Expec{(X-\Expec{X|Y} + \Expec{X|Y} -\hat{X}(Y))^2}\\
    =& \Expec{(X-\Expec{X|Y})^2} + \Expec{2(X-\Expec{X|Y})(\Expec{X|Y}-\hat{X}(Y))}+\Expec{(\Expec{X|Y} -\hat{X}(Y))^2}
\end{align}
Consider the second term.
Due to the tower property, we have
\begin{align}
\Expec{2(X-\Expec{X|Y})(\Expec{X|Y}-\hat{X}(Y))}=& \Esub{Y}{2\Esub{X}{(X-\Expec{X|Y})(\Expec{X|Y}-\hat{X}(Y))|Y}}\\
=& \Esub{Y}{2(\Expec{X|Y}-\hat{X}(Y))\Esub{X}{(X-\Expec{X|Y})|Y}}\\
=& 0.
\end{align}
The last inequality is due to $ \Expec{X - \Expec{X|Y}|Y} = \Expec{X|Y} - \Expec{X|Y}=0$.
Thus,  
\begin{align}
    \Expec{(X-\hat{X}(Y))^2}=& \Expec{(X-\Expec{X|Y})^2} +\Expec{(\Expec{X|Y} -\hat{X}(Y))^2}\\
    \geq& \Expec{(X-\Expec{X|Y})^2}.
\end{align}
Which implies $\hat{X}(Y) = \Expec{X|Y}$ is the MMSE.
\end{proof}

\begin{definition}
    Let $X_1, X_2$ be a jointly Gaussian with mean $\bmu=(\mu_1, \mu_2)$ and covariance matrix
    \begin{align}
        \Sigma = \begin{bmatrix}
            \sigma_1^2 &\rho \sigma_1\sigma_2\\
            \rho\sigma_1\sigma_2 & \sigma_2^2
        \end{bmatrix}.
    \end{align}
    We say $(X_1, X_2)\sim\cN({\bmu}, \Sigma)$ and the corresponding joint probability density function (pdf)
    \begin{align}
        f_{X_1, X_2}(x_1, x_2) = \frac{1}{\sqrt{(2\pi)^2 \det(\Sigma)}} \exp\left(-\frac{1}{2}(x^2-\bmu)\Sigma^{-1} (x^2-\bmu)\right).
    \end{align}
    where $x^2 = (x_1, x_2)$.
\end{definition}

\begin{theorem}
    If $(X_1, X_2)\sim \cN(\bmu, \Sigma)$, then $X_1|X_2$ is also Gaussian with mean $\mu_1 + \frac{\sigma_{12}}{\sigma_{22}}(x_2-\mu_2)$
    and variance $\sigma_{22} - \frac{\sigma_{12}\sigma_{21}}{\sigma_{22}}$ where 
    \begin{align}
        \Sigma = \begin{bmatrix}
            \sigma_{11} & 
            \sigma_{12} \\ 
            \sigma_{21} & 
            \sigma_{22}  
        \end{bmatrix}
    \end{align}
\end{theorem}

\subsection{Positive Definite Matrices}

\begin{definition}
Let $A$ be a real symmetric $n \times n$ matrix.
\begin{enumerate}
    \item $A$ is \emph{positive definite}, denoted $A > 0$, if
    \[
        x^{T} A x > 0, \quad \forall x \in \mathbb{R}^n, \, x \neq 0.
    \]
    \item $A$ is \emph{nonnegative definite} (or positive semi definite), denoted $A \ge 0$, if
    \[
        x^{T} A x \ge 0, \quad \forall x \in \mathbb{R}^n.
    \]
\end{enumerate}
\end{definition}

It is known from basic linear algebra that:
\begin{enumerate}
    \item A symmetric matrix has $n$ real eigenvalues.
    \item If $A > 0$, all eigenvalues are positive.
    \item If $A \ge 0$, all eigenvalues are nonnegative.
    \item Eigenvectors corresponding to distinct eigenvalues are orthogonal, and
    can be chosen to form an orthonormal basis of $\mathbb{R}^n$.
\end{enumerate}

Every symmetric matrix $A$ can be written as
\[
A = \sum_{i=1}^{n} \lambda_i z_i z_i^T,
\]
where $\lambda_i$ are eigenvalues and $z_i$ are orthonormal eigenvectors (which is called spectral decomposition).
For $A \ge 0$, all $\lambda_i \ge 0$, allowing us to define the symmetric
square root
\[
B = \sum_{i=1}^{n} \sqrt{\lambda_i} z_i z_i^T, \quad \text{so that } \; B^2 = A.
\]
If $A > 0$, then all $\lambda_i > 0$, and we can define the inverse as
\[
A^{-1} = \sum_{i=1}^{n} \frac{1}{\lambda_i} z_i z_i^T.
\]


\begin{exercise}
    For positive semidefinite matrix $A$, show that there exists a matrix $B$ such that $A = B^TB$.
\end{exercise}

\subsection{Multivariate Gaussian}

Our interest in positive definite matrices stems from their role in defining multivariate normal densities.
When $A > 0$, the quadratic form $q(x) = x^T A x$ grows unbounded as $\|x\| \to \infty$,
and $e^{-q(x)}$ decays to zero, thus enabling valid probability densities.

There are three equivalent ways to define the multivariate normal distribution.

\begin{definition}[Density definition]\label{def:MV1}
A random vector $X \in \mathbb{R}^n$ has a (nondegenerate) \emph{multivariate normal distribution} if its joint PDF is
\[
f_X(x) = \frac{1}{(2\pi)^{n/2} |V|^{1/2}}
\exp\left( -\frac{1}{2} (x - \mu)^T V^{-1} (x - \mu) \right),
\]
for some mean vector $\mu \in \mathbb{R}^n$ and positive definite covariance matrix $V$.
\end{definition}

\begin{definition}[Constructive definition]\label{def:MV2}
A random vector $X$ is multivariate normal if
\[
X = D W + \mu,
\]
where $W$ is a vector of independent standard normal random variables $N(0,1)$,
and $D$ is a real matrix.
\end{definition}

\begin{definition}[Projection definition]\label{def:MV3}
A random vector $X$ is multivariate normal if, for every real vector $a$,
the scalar random variable $a^T X$ is (univariate) normal.
\end{definition}

\paragraph{Degeneracy.}
The first definition requires a PDF, which only exists if $V$ is positive definite.
When $V$ is singular (e.g., $X_2 = 0$ always), the distribution is called
\emph{degenerate}, as all mass is concentrated on a lower-dimensional subspace.
Definitions~\ref{def:MV2} and Definition~\ref{def:MV3} remain valid for degenerate normals.



%%% Means and COVs
\subsection{Means and Covariances of Vector Random Variables}

Let $X = (X_1, \dots, X_n)^T$ be a random vector.  
Its mean vector is
\[
E[X] = \begin{bmatrix} E[X_1] \\ \vdots \\ E[X_n] \end{bmatrix}.
\]
Given random vectors $X \in \mathbb{R}^n$ and $Y \in \mathbb{R}^m$, the
\emph{covariance matrix} is
\[
\mathrm{Cov}(X, Y) = E\left[(X - E[X])(Y - E[Y])^T\right].
\]
In particular, $\mathrm{Cov}(X, X)$ is an $n \times n$ symmetric matrix.

\begin{exercise}
Show that $\mathrm{Cov}(X, X)$ is nonnegative definite.
\end{exercise}

For scalar random variables $X_i$ and $X_j$, the \emph{correlation coefficient} is defined as
\[
\rho_{ij} = \frac{\mathrm{Cov}(X_i, X_j)}
{\sqrt{\mathrm{Var}(X_i)\,\mathrm{Var}(X_j)}}.
\]
The correlation coefficient $\rho_{ij}$ satisfies $-1 \le \rho_{ij} \le 1$,  
and provides a normalized measure of the linear dependence between $X_i$ and $X_j$.
If $\rho_{ij}=0$, then we say random variables $X_i$ and $X_j$ are uncorrelated.






%%% Multivariate Normal
\subsection{Multivariate Normal}
\begin{theorem}[Properties of multivariate normal distributions]
Let $X = D W + \mu$, where $W$ has independent $N(0,1)$ components.
Then:
\begin{enumerate}
    \item Each component $X_i$ is univariate normal with mean $\mu_i$.
    \item $\mathrm{Cov}(X, X) = D D^T$.
    \item If $Y = C X + d$, where $C$ is an $m \times n$ matrix and $d \in \mathbb{R}^m$, then
    \[
        Y \sim \mathcal{N}(C\mu + d, \; C D D^T C^T).
    \]
    \item If $D$ is invertible, then $X$ is nondegenerate with covariance $V = D D^T$,
    and has the density in Definition 2.
    \item The joint CDF of $X$ is completely determined by its mean $\mu$ and covariance $V$.
    \item The components of $X$ are uncorrelated if and only if they are independent.
    \item If
    \[
    \begin{bmatrix} X \\ Y \end{bmatrix}
    \sim \mathcal{N} \left(
    \begin{bmatrix} \mu_X \\ \mu_Y \end{bmatrix},
    \begin{bmatrix}
        V_{XX} & V_{XY} \\
        V_{YX} & V_{YY}
    \end{bmatrix}
    \right),
    \quad V_{YY} > 0,
    \]
    then:
    \begin{enumerate}
        \item $E[X \mid Y] = \mu_X + V_{XY} V_{YY}^{-1} (Y - \mu_Y)$.
        \item Let $\tilde{X} = X - E[X \mid Y]$. Then $\tilde{X}$ is independent of $Y$ and of $E[X \mid Y]$.
        \item $\mathrm{Cov}(\tilde{X}, \tilde{X}) = V_{XX} - V_{XY} V_{YY}^{-1} V_{YX}$.
    \end{enumerate}
\end{enumerate}
\end{theorem}

\begin{proof}
\begin{enumerate}
\item 
Under Definition~\ref{def:MV3}, each component $X_i$ is a linear function of independent normal random variables, and hence is itself normal.  
Since $E[W] = 0$, we have
\[
E[X_i] = E[e_i^T(DW + \mu)] = e_i^T \mu = \mu_i.
\]

\item 
For simplicity, consider the zero-mean case ($\mu = 0$).  
Then
\begin{align*}
\mathrm{Cov}(X, X) 
&= E[XX^T] 
= E[DWW^T D^T] 
= D\,E[WW^T]\,D^T 
= D D^T,
\end{align*}
since the components of $W$ are independent with unit variance, i.e., $E[WW^T] = I$.

\item 
Let $Y = C X + d = C(DW + \mu) + d$.  
Then $Y$ is a linear function of independent $N(0,1)$ variables, hence multivariate normal.  
We have
\[
E[Y] = C \mu + d, \qquad 
\mathrm{Cov}(Y, Y) = C\,\mathrm{Cov}(X, X)\,C^T = C D D^T C^T.
\]

\item 
This part follows from a standard result on transformations of random variables.  
For simplicity, take $\mu = 0$.  
When $X = D W$ with $D$ invertible, the change of variables formula gives
\[
f_X(x) = \frac{f_W(D^{-1}x)}{|\det D|}.
\]
Since the components of $W$ are i.i.d.\ $N(0,1)$, we have
\[
f_W(w) = \frac{1}{(2\pi)^{n/2}} 
\exp\!\left(-\frac{1}{2} w^T w \right).
\]
Substituting $w = D^{-1}x$, we obtain
\[
f_X(x) = \frac{1}{(2\pi)^{n/2} |D D^T|^{1/2}}
\exp\!\left(-\frac{1}{2} x^T (D^{-1})^T D^{-1} x \right).
\]
This is exactly the form in Definition~2 with $V = D D^T$.  
By part~(b), $\mathrm{Cov}(X, X) = V$.  
The same argument holds for $\mu \ne 0$ by replacing $x$ with $(x - \mu)$.

\item 
From part~(d), the joint PDF of $X$ is determined by $\mu$ and $V = D D^T$.
In the degenerate case (when $V$ is singular), there is no joint PDF, but the distribution is still determined by $\mu$ and $V$.
To see this, consider the moment generating function (MGF)
\[
M_X(s) = E[e^{s^T X}].
\]
Since $s^T X$ is a scalar normal variable with mean $s^T \mu$ and variance $s^T V s$, we have
\[
M_X(s) = \exp\!\left(s^T \mu + \frac{1}{2} s^T V s\right).
\]
By the inversion theorem for transforms, the distribution of $X$ is completely determined by $\mu$ and $V$.

\item 
If the components of $X$ are independent, then they are necessarily uncorrelated.
For the converse, assume the components are uncorrelated so that $V$ is diagonal.
Let $Y$ be a random vector whose components are independent normal variables with the same means and variances as $X$.
Then $X$ and $Y$ have identical means and covariances; hence by part~(e), they have the same distribution.
Therefore, the components of $X$ are also independent.

Alternatively, when $V$ is invertible and diagonal, the joint density factorizes as
\[
f_X(x) = \frac{1}{(2\pi)^{n/2} |V|^{1/2}} 
\exp\!\left(-\frac{1}{2} (x - \mu)^T V^{-1} (x - \mu) \right)
= \prod_{i=1}^{n} f_{X_i}(x_i),
\]
confirming independence.

\item 
For simplicity, assume zero mean.  
Define the random vector
\[
\hat{X} = V_{XY} V_{YY}^{-1} Y.
\]
Then
\[
E[X Y^T] = V_{XY} = E[\hat{X} Y^T],
\]
implying $E[(X - \hat{X}) Y^T] = 0$, i.e., $X - \hat{X}$ is uncorrelated with $Y$.
Since $(X, Y)$ is jointly normal, $(X - \hat{X}, Y)$ is also multivariate normal,
and uncorrelatedness implies independence.  
Therefore, $X - \hat{X}$ is independent of $Y$.

Now, for any measurable function $g$,
\[
E[(X - \hat{X}) g(Y)] = 0,
\]
which by the definition of conditional expectation implies $\hat{X} = E[X \mid Y]$.

Thus,
\[
E[X \mid Y] = V_{XY} V_{YY}^{-1} Y.
\]

For part~(ii), define $\tilde{X} = X - E[X \mid Y]$.  
Since $\tilde{X}$ is independent of $Y$, it is also independent of $E[X \mid Y]$.

For part~(iii),
\begin{align*}
\mathrm{Cov}(\tilde{X}, \tilde{X})
&= E[\tilde{X} \tilde{X}^T]
= E[(X - \hat{X})(X - \hat{X})^T] \\
&= E[XX^T] - E[X \hat{X}^T] - E[\hat{X} X^T] + E[\hat{X} \hat{X}^T] \\
&= V_{XX} - V_{XY} V_{YY}^{-1} V_{YX}.
\end{align*}
This covariance is also equal to $\mathrm{Cov}(\tilde{X}, \tilde{X} \mid Y)$
since $\tilde{X}$ and $Y$ are independent.
\end{enumerate}
\end{proof}



%%% Gaussian Channel
\subsection{Gaussian Channel}
Let $X \sim \mathcal{N}(0, \sigma_x^2)$ and $Z \sim \mathcal{N}(0, \sigma_z^2)$ be independent random variables, and observe
\[
Y = X + Z.
\]
Then $(X, Y)$ is jointly Gaussian with
\[
\mathrm{Var}(X) = \sigma_x^2, \quad 
\mathrm{Var}(Y) = \sigma_x^2 + \sigma_z^2, \quad
\mathrm{Cov}(X, Y) = \sigma_x^2.
\]

Because $(X, Y)$ is jointly Gaussian,
\[
X \mid Y = y \sim \mathcal{N}\big( \mu_{X\mid Y}(y), \, \sigma^2_{X\mid Y} \big),
\]
where
\[
\mu_{X\mid Y}(y)
= \frac{\mathrm{Cov}(X,Y)}{\mathrm{Var}(Y)}\, y
= \frac{\sigma_x^2}{\sigma_x^2 + \sigma_z^2}\, y,
\qquad
\sigma^2_{X\mid Y}
= \sigma_x^2 - \frac{\mathrm{Cov}(X,Y)^2}{\mathrm{Var}(Y)}
= \frac{\sigma_x^2 \sigma_z^2}{\sigma_x^2 + \sigma_z^2}.
\]
Hence, the MMSE estimator is the posterior mean:
\[
\widehat{X}_{\mathrm{MMSE}}(Y)
= \mathbb{E}[X \mid Y]
= \frac{\sigma_x^2}{\sigma_x^2 + \sigma_z^2} \, Y.
\]

Note that the followings are two equivalent sampling procedures for $(X, Y)$.
\begin{enumerate}
    \item \textbf{Forward (prior + noise):}  
    Sample $X \sim \mathcal{N}(0, \sigma_x^2)$ and $Z \sim \mathcal{N}(0, \sigma_z^2)$ independently, then set
    \[
    Y = X + Z.
    \]
    
    \item \textbf{Observation-first (posterior decomposition):}  
    First sample 
    \[
    Y \sim \mathcal{N}(0, \sigma_x^2 + \sigma_z^2),
    \]
    then, conditional on $Y$, sample
    \[
    X = \frac{\sigma_x^2}{\sigma_x^2 + \sigma_z^2}\, Y
      + \sqrt{\frac{\sigma_x^2 \sigma_z^2}{\sigma_x^2 + \sigma_z^2}}\, W,
    \quad W \sim \mathcal{N}(0,1)
    \text{ independent of } Y.
    \]
\end{enumerate}
Equivalently, $X \mid Y$ can be viewed as adding independent Gaussian noise with variance 
$\frac{\sigma_x^2 \sigma_z^2}{\sigma_x^2 + \sigma_z^2}$ 
to the MMSE estimate $\widehat{X}_{\mathrm{MMSE}}(Y)$.

We can do the simple sanity check.
As $\sigma_z^2 \to 0$, we have $\mu_{X\mid Y} \to Y$ and $\sigma^2_{X\mid Y} \to 0$.  
As $\sigma_z^2 \to \infty$, we have $\mu_{X\mid Y} \to 0$ and $\sigma^2_{X\mid Y} \to \sigma_x^2$.



%%% Diffusion
\subsection{Gaussian Diffusion}
Let $\{X_n\}_{n\ge0}$ be a Gaussian process defined by the recursion
\begin{align}
    X_n = \sqrt{1 - \beta}\, X_{n-1} + \sqrt{\beta}\, Z_n, \qquad 
    Z_n \sim \mathcal{N}(0,1),
\end{align}
where each $Z_n$ is an independent standard normal noise.
Equivalently,
\[
X_n \mid X_{n-1} \sim \mathcal{N}\!\left(\sqrt{1-\beta}\,X_{n-1},\, \beta\right).
\]

Suppose the initial state $X_0 \sim f_0$ has mean $0$ and variance $1$.  
Then the first and second moments of $\{X_n\}$ satisfy
\begin{align}
    \mathbb{E}[X_n] 
    &= \sqrt{1-\beta}\,\mathbb{E}[X_{n-1}] + \sqrt{\beta}\,\mathbb{E}[Z_n]
    = 0,\\
    \mathrm{Var}(X_n) 
    &= (1-\beta)\,\mathrm{Var}(X_{n-1}) + \beta\,\mathrm{Var}(Z_n)
    = 1,
\end{align}
for all $n \ge 1$. Hence, $\{X_n\}$ remains zero-mean and unit-variance for all steps.

\begin{exercise}
    What is the limiting distribution of $X_n$ as $n \to \infty$?
\end{exercise}

Let $M_{X_n}(t)=\mathbb{E}[e^{tX_n}]$ denote the moment generating function (MGF) of $X_n$.
Then
\begin{align}
    M_{X_n}(t)
    &= M_{X_{n-1}}\!\left(\sqrt{1-\beta}\,t\right)
       e^{\frac{\beta t^2}{2}},\\
    &= M_{X_{n-2}}\!\left((\sqrt{1-\beta})^2 t\right)
       e^{\frac{(1-(1-\beta))t^2}{2}},\\
    &\;\;\vdots\\
    &= M_{X_0}\!\left((\sqrt{1-\beta})^n t\right)
       \exp\!\left(\tfrac{1 - (1-\beta)^n}{2}\,t^2\right).
\end{align}
As $n \to \infty$, $(\sqrt{1-\beta})^n \to 0$, so
\[
M_{X_n}(t) \to e^{t^2/2},
\]
which is the MGF of $\mathcal{N}(0,1)$.  
Hence,
\[
X_n \xrightarrow{d} \mathcal{N}(0,1),
\]
i.e., the process converges to the standard normal distribution regardless of the initial $f_0$.

When $\beta$ is small, the reverse conditional distribution can be locally approximated by a Gaussian:
\[
X_{n-1} \mid X_n 
\;\approx\; \mathcal{N}\!\big(\mu_{n-1\mid n},\, \sigma_{n-1\mid n}^2\big),
\]
where the parameters $(\mu_{n-1\mid n}, \sigma_{n-1\mid n}^2)$ can be derived from the joint Gaussian form of $(X_{n-1}, X_n)$.

\bigskip

Let $\{\mathbf{X}_n\}_{n\ge0}$ be a sequence of random vectors in $\mathbb{R}^d$ defined by
\begin{align}
    \mathbf{X}_n 
    = \sqrt{1-\beta}\, \mathbf{X}_{n-1} 
    + \sqrt{\beta}\, \mathbf{Z}_n,
    \qquad \mathbf{Z}_n \sim \mathcal{N}(\mathbf{0}, \mathbf{I}_d),
\end{align}
with $\mathbf{X}_0 \sim f_0(\mathbf{x})$ having mean $\mathbf{0}$ and covariance $\mathbf{I}_d$.
Then
\[
\mathbf{X}_n \mid \mathbf{X}_{n-1} 
\sim \mathcal{N}\!\left(
    \sqrt{1-\beta}\,\mathbf{X}_{n-1},\,
    \beta\,\mathbf{I}_d
\right),
\]
and the marginal statistics remain stationary:
\[
\mathbb{E}[\mathbf{X}_n] = \mathbf{0},
\qquad
\mathrm{Cov}(\mathbf{X}_n) = \mathbf{I}_d.
\]
The limiting distribution satisfies
\[
\mathbf{X}_n \xrightarrow{d} \mathcal{N}(\mathbf{0}, \mathbf{I}_d),
\quad \text{as } n \to \infty.
\]

Now let's consider the multidimensional case, which includes the most practical scenarios such as image, audio, and video generations.
As in the scalar case, when $\beta$ is small, the reverse conditional can be approximated by a Gaussian:
\[
\mathbf{X}_{n-1} \mid \mathbf{X}_n 
\;\approx\;
\mathcal{N}\!\big(
\boldsymbol{\mu}_{n-1\mid n},
\boldsymbol{\Sigma}_{n-1\mid n}
\big),
\]
where $\boldsymbol{\mu}_{n-1\mid n},  \boldsymbol{\Sigma}_{n-1\mid n}$ can be approximated via Neural Networks.
This Gaussian approximation underlies the reverse diffusion formulation used in diffusion probabilistic models.
That is, each backward step can be obtained by sampling according to the above reverse processes.

Based on this forward process and (approximated) reverse process, now we are ready to generate a data similar to the data distribution.
\begin{enumerate}
    \item[\textbf{1.}] 
    Initialize the terminal state from the stationary (noise) distribution (for large enough $T$):
    \[
        \mathbf{X}_T \sim \mathcal{N}(\mathbf{0}, \mathbf{I}_d).
    \]

    \item[\textbf{2.}]
    For $n = T, T-1, \dots, 1$, sample $\mathbf{X}_{n-1}$ according to
    \begin{align*}
\mathbf{X}_{n-1} \mid \mathbf{X}_n  \;\approx\; \mathcal{N}\!\big( \boldsymbol{\mu}_{n-1\mid n}, \boldsymbol{\Sigma}_{n-1\mid n} \big),
    \end{align*}
    \item[\textbf{3.}]
    The resulting $\mathbf{X}_0$ is a sample from the reversed process,
    which is (approximately) equivalent to sample according to the data distribution.
\end{enumerate}

\section{Gaussian Discriminant}

We consider binary classification with class-conditional multivariate Gaussians:
\[
X \mid Y=0 \sim \mathcal{N}(\boldsymbol{\mu}_0,\Sigma_0),
\qquad
X \mid Y=1 \sim \mathcal{N}(\boldsymbol{\mu}_1,\Sigma_1),
\]
where $X\in\mathbb{R}^n$. The MLE decision rule (equivalently, MAP with equal priors) picks $y=0$ if
\begin{align}
\frac{1}{\sqrt{(2\pi)^n |\Sigma_0|}}
\exp\!\left(
-\tfrac{1}{2}(x-\boldsymbol{\mu}_0)^\top \Sigma_0^{-1} (x-\boldsymbol{\mu}_0)
\right)
>
\frac{1}{\sqrt{(2\pi)^n |\Sigma_1|}}
\exp\!\left(
-\tfrac{1}{2}(x-\boldsymbol{\mu}_1)^\top \Sigma_1^{-1} (x-\boldsymbol{\mu}_1)
\right).
\end{align}
With non-equal priors $\pi_y = \mathbb{P}(Y=y)$, the MAP rule multiplies each side by $\pi_0$ and $\pi_1$, respectively.

Both MLE and MAP reqwuires the knowledge of parameters $\mathbf{\mu}$ and $\mathbf{\Sigma}$'s,
which requires the MLE parameter estimation.
Given i.i.d.\ samples $\{x^{(i)}\}_{i=1}^N$ from a single Gaussian class $\mathcal{N}(\boldsymbol{\mu},\Sigma)$, the log-likelihood is
\begin{align}
\ell(\boldsymbol{\mu},\Sigma)
&= \sum_{i=1}^N \log p(x^{(i)};\boldsymbol{\mu},\Sigma)
= -\frac{N n}{2}\log(2\pi) - \frac{N}{2}\log|\Sigma|
  - \frac{1}{2}\sum_{i=1}^N (x^{(i)}-\boldsymbol{\mu})^\top \Sigma^{-1} (x^{(i)}-\boldsymbol{\mu}).
\end{align}

Taking the gradient with respect to $\boldsymbol{\mu}$:
\begin{align}
\nabla_{\boldsymbol{\mu}} \ell
= \sum_{i=1}^N \Sigma^{-1}(x^{(i)}-\boldsymbol{\mu})
= \Sigma^{-1}\!\left(\sum_{i=1}^N x^{(i)} - N\boldsymbol{\mu}\right).
\end{align}
Setting $\nabla_{\boldsymbol{\mu}}\ell=0$ yields the MLE
\[
\widehat{\boldsymbol{\mu}}=\frac{1}{N}\sum_{i=1}^N x^{(i)}.
\]

For the convaraince matrix estimation, it is convenient to reparameterize with the precision matrix $\Phi:=\Sigma^{-1}$. Using $|\Sigma|=|\Phi^{-1}|=|\Phi|^{-1}$, we have
\begin{align}
\ell(\boldsymbol{\mu},\Phi)
= -\frac{N n}{2}\log(2\pi) + \frac{N}{2}\log|\Phi|
  - \frac{1}{2}\sum_{i=1}^N (x^{(i)}-\boldsymbol{\mu})^\top \Phi (x^{(i)}-\boldsymbol{\mu}).
\end{align}
Define the scatter matrix
\[
S := \sum_{i=1}^N (x^{(i)}-\boldsymbol{\mu})(x^{(i)}-\boldsymbol{\mu})^\top.
\]
Using the trace trick,
\[
(x-\boldsymbol{\mu})^\top \Phi (x-\boldsymbol{\mu})
= \mathrm{tr}\!\left( (x-\boldsymbol{\mu})(x-\boldsymbol{\mu})^\top \Phi \right),
\]
we obtain the gradients (matrix identities recalled below):
\begin{align}
\nabla_{\Phi}\ell
= \frac{N}{2} \, \Phi^{-T} - \frac{1}{2} S.
\end{align}
Setting $\nabla_{\Phi}\ell=0$ gives
\[
\Phi^{-T} = \frac{1}{N} S
\quad\Longrightarrow\quad
\widehat{\Sigma} = \Phi^{-1} = \frac{1}{N} S
= \frac{1}{N} \sum_{i=1}^N (x^{(i)}-\widehat{\boldsymbol{\mu}})(x^{(i)}-\widehat{\boldsymbol{\mu}})^\top.
\]
\emph{Remark.} This is the MLE (biased) covariance. 
The unbiased sample covariance replaces $1/N$ with $1/(N-1)$.

In the above derivation, the following matrix-calculus identities were used.
For matrices of compatible sizes:
\begin{align}
&\nabla_{X} \,\mathrm{tr}(A^\top X) = A, \qquad
\nabla_{X} \,\mathrm{tr}(A X) = A^\top,\\
&\nabla_{X} \log|X| = X^{-T} \quad (\text{$X$ invertible}).
\end{align}
Hence, for any fixed $A$, $\nabla_{\Phi}\mathrm{tr}(A\Phi)=A^\top$ and
$\nabla_{\Phi}\log|\Phi|=\Phi^{-T}$.


\begin{exercise}
For a matrix $A$ and a positive semidefinite matrix $\Phi$, show that
$\nabla_{\Phi}\,\mathrm{tr}(A\Phi)=A^\top$.
\end{exercise}

\begin{exercise}
Show that for an invertible (symmetric) positive definite $\Phi$,
$\nabla_{\Phi}\,\log|\Phi|=\Phi^{-T}=\Phi^{-1}$.
\end{exercise}
