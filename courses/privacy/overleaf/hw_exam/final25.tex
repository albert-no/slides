\item {\bf Nonuniform Time Scale Diffusion} {(\it 30 points)}\\
For a data variable $X$, consider a diffusion model with latent variables $Z_t$,
where $t$ runs from $t=0$ (least noisy) to $t=1$ (most noisy).
The distribution of latent variable $Z_t$ conditioned on $X$ for $t\in[0, 1]$ is given by
\begin{align*}
    q(Z_t|X)\sim \mathcal{N}(\alpha_t X, \sigma_t^2).
\end{align*}
Furthermore, for $t>s$, the conditional distribution $Z_t$ given $Z_s$ is given by
\begin{align*}
    q(Z_t|Z_s) \sim \mathcal{N}(\alpha_{t|s} Z_s, \sigma_{t|s}^2)
\end{align*}
where $\alpha_{t|s} = \alpha_t/\alpha_s$ and $\sigma_{t|s}^2 = \sigma_t^2 - \alpha_{t|s}^2\sigma_s^2$.
\begin{enumerate}
\item Let $q(Z_s|Z_t, X) \sim \mathcal{N}(\mu_Q(Z_t, X;s, t), \sigma_Q^2(s, t))$.
Find $\mu_Q(Z_t, X;s, t)$ and $\sigma_Q^2(s, t)$.  {(\it 10 points)}
\item Let $t_i = i/T$ for $0\leq i\leq T$.
Consider a following reverse distribution $p$:
\begin{align*}
    p(X) = \int_{\mathbf{Z}} p(Z_1) p(X|Z_0) \prod_{i=1}^T p(Z_{t_{i-1}}|Z_{t_i})\, d\mathbf{Z}
\end{align*}
where $Z$ represents the whole $(Z_{t_0}, \dots, Z_{t_T})$.

Show the following variational bound
\begin{align*}
    -\log p(X) \leq D(q(Z_1|X)\| p(Z_1)) + \field{E}_{q(Z_0|X)}[-\log p(X|Z_0)] + \mathcal{L}(X)
\end{align*}
where
\begin{align*}
    \mathcal{L}(X) = \sum_{i=1}^T \field{E}_{q(Z_{t_i}|X)} \left[D(q(Z_{t_{i-1}}|Z_{t_i}, X)\| p(Z_{t_{i-1}}|Z_{t_i})\right].
\end{align*}
{(\it 10 points)}
\item Let $p(Z_{t_{i-1}}|Z_{t_i}) = q(Z_{t_{i-1}}|Z_{t_i}, X=\hat{x}_{\theta}(Z_{t_i}; t_i))$.
Show that 
\begin{align*}
    \mathcal{L}(X) = \frac{1}{2} \sum_{i=1}^T \field{E}_{q(Z_{t_i}|X)} \left[\left(\mbox{SNR}(t_{i-1}) - \mbox{SNR}(t_i)\right) \|X - \hat{x}_\theta(Z_{t_i};t_i)\|^2\right]
\end{align*}
where $\mbox{SNR}(t) = \alpha_t^2 / \sigma_t^2$.
{(\it 10 points)}
\end{enumerate}

\vskip .2in
\begin{solution}{\bf Solution: Nonuniform Time Scale Diffusion.}
\begin{enumerate}
    \item Since \[
    \left.\begin{bmatrix}
    Z_s\\Z_t
    \end{bmatrix}\right\vert X \sim \mathcal{N}\left(\begin{bmatrix}
    \alpha_sX\\\alpha_tX
    \end{bmatrix},\,\begin{bmatrix}
    \sigma_s^2&\alpha_{t|s}\sigma_s^2\\\alpha_{t|s}\sigma_s^2&\sigma_t^2
    \end{bmatrix}\right),
    \]
    the conditional distribution is given by \[Z_s\,|\,Z_t,X\sim \mathcal{N}\left(\alpha_sX+\frac{\alpha_{t|s}\sigma_s^2}{\sigma_t^2}(Z_t-\alpha_tX),\,\sigma_s^2-\frac{(\alpha_{t|s}\sigma_s^2)^2}{\sigma_t^2}\right).\]
    Thus,
    \[\begin{aligned}\mu_Q(Z_t,X;s,t)&=\alpha_sX+\frac{\alpha_{t|s}\sigma_s^2}{\sigma_t^2}(Z_t-\alpha_tX)\\&=\frac{\alpha_{t|s}\sigma_s^2}{\sigma_t^2}Z_t+\alpha_sX-\frac{\alpha_{t|s}\sigma_s^2}{\sigma_t^2}(\alpha_{t|s}\alpha_s)X\\&=\frac{\alpha_{t|s}\sigma_s^2}{\sigma_t^2}Z_t+\alpha_s\left(1-\frac{\alpha_{t|s}^2\sigma_s^2}{\sigma_t^2}\right)X\\&=\frac{\alpha_{t|s}\sigma_s^2}{\sigma_t^2}Z_t+\frac{\alpha_s\sigma_{t|s}^2}{\sigma_t^2}X,
    \end{aligned}
    \]
    and
    \[
    \sigma_Q^2(s,t)=\sigma_s^2-\frac{(\alpha_{t|s}\sigma_s^2)^2}{\sigma_t^2}=\frac{\sigma_s^2\sigma_{t|s}^2}{\sigma_t^2}.
    \]
    \item First, we have
    \[\begin{aligned}
        -\log p(X)&=-\log\int p(X,\mathbf{Z})d\mathbf{Z}\\&=-\log\int q(\mathbf{Z}|X)\frac{p(X,\mathbf{Z})}{q(\mathbf{Z}|X)}d\mathbf{Z}\\&=-\log\mathbb{E}_{q(\mathbf{Z}|X)}\left[\frac{p(X,\mathbf{Z})}{q(\mathbf{Z}|X)}\right]\\&\leq\mathbb{E}_{q(\mathbf{Z}|X)}\left[\log\frac{q(\mathbf{Z}|X)}{p(X,\mathbf{Z})}\right]
    \end{aligned}\]
    by Jensen's inequality.
    Now we have factorizations
    \[p(X, \mathbf{Z}) =p(X,Z_{t_0},\dots, Z_{t_T})=p(Z_1) p(X|Z_0) \prod_{i=1}^{T} p(Z_{t_{i-1}}|Z_{t_i})
    \]
    and
    \[
    q(\mathbf{Z}|X) = q(Z_{t_0}, \dots, Z_{t_T}|X) = q(Z_0|X) \prod_{i=1}^{T} q(Z_{t_i}|Z_{t_{i-1}},X).
    \]
    By Bayes' theorem,
    \[q(Z_{t_i}|Z_{t_{i-1}},X)=q(Z_{t_i}|Z_{t_{i-1}})=\frac{q(Z_{t_{i-1}}|Z_{t_i},X)q(Z_{t_i}|X)}{q(Z_{t_{i-1}}|X)}.\]
    Then,
    \[\begin{aligned}
    &\mathbb{E}_{q(\mathbf{Z}|X)}\left[\log\frac{q(\mathbf{Z}|X)}{p(X,\mathbf{Z})}\right] \\
    &= \mathbb{E}_{q(\mathbf{Z}|X)}\left[\log\frac{q(Z_0|X) \prod_{i=1}^{T} q(Z_{t_i}|Z_{t_{i-1}})}{p(Z_1) p(X|Z_0) \prod_{i=1}^{T} p(Z_{t_{i-1}}|Z_{t_i})}\right] \\
    &= \mathbb{E}_{q(\mathbf{Z}|X)}\left[ \log q(Z_0|X) - \log p(X|Z_0) - \log p(Z_1) \right. \\
    & \qquad \left. + \sum_{i=1}^{T} \left( \log q(Z_{t_i}|Z_{t_{i-1}}) - \log p(Z_{t_{i-1}}|Z_{t_i}) \right) \right] \\
    &= \mathbb{E}_{q(\mathbf{Z}|X)}\left[ \log q(Z_0|X) - \log p(X|Z_0) - \log p(Z_1) \right. \\
    & \qquad \left. + \sum_{i=1}^{T} \left( \log\frac{q(Z_{t_{i-1}}|Z_{t_i}, X)q(Z_{t_i}|X)}{q(Z_{t_{i-1}}|X)} - \log p(Z_{t_{i-1}}|Z_{t_i}) \right) \right] \\
    &= \mathbb{E}_{q(\mathbf{Z}|X)}\left[ \log q(Z_0|X) - \log p(X|Z_0) - \log p(Z_1) \right. \\
    & \qquad \left. + \sum_{i=1}^{T} \left( \log q(Z_{t_i}|X) - \log q(Z_{t_{i-1}}|X) \right) + \sum_{i=1}^{T} \log\frac{q(Z_{t_{i-1}}|Z_{t_i}, X)}{p(Z_{t_{i-1}}|Z_{t_i})} \right] \\
    &= \mathbb{E}_{q(\mathbf{Z}|X)}\left[ \log q(Z_0|X) - \log p(X|Z_0) - \log p(Z_1) \right. \\
    & \qquad \left. + (\log q(Z_1|X) - \log q(Z_0|X)) + \sum_{i=1}^{T} \log\frac{q(Z_{t_{i-1}}|Z_{t_i}, X)}{p(Z_{t_{i-1}}|Z_{t_i})} \right] \\
    &=\mathbb{E}_{q(Z_0|X)}[-\log p(X|Z_0)] + D(q(Z_1|X) \| p(Z_1)) + \mathcal{L}(X).
    \end{aligned}\]
    \item We have \[p(Z_{t_{i-1}}|Z_{t_i})\sim \mathcal{N}(\mu_{\theta}(Z_{t_i};t_{i-1},t_i),\,\sigma_Q^2(t_{i-1},t_i)),\] where $\mu_{\theta}(Z_{t_i};t_{i-1},t_i) = \frac{\alpha_{t_i|t_{i-1}}\sigma_{t_{i-1}}^2}{\sigma_{t_i}^2}Z_t+\frac{\alpha_{t_{i-1}}\sigma_{t_i|t_{i-1}}^2}{\sigma_{t_i}^2}\hat{x}_{\theta}(Z_{t_i};t_i)$.
    The KL divergence between two Gaussian distributions with common variance is given by:
    \[
    \begin{aligned}
        D(q(Z_{t_{i-1}}|Z_{t_i}, X)\| p(Z_{t_{i-1}}|Z_{t_i}))&=\frac{1}{2\sigma^2_Q(t_{i-1},t_i)}\|\mu_Q-\mu_{\theta}\|^2\\
        &=\frac{\sigma_{t_i}^2}{2\sigma^2_{t_i|t_{i-1}}\sigma^2_{t_{i-1}}}\frac{\alpha^2_{t_{i-1}}\sigma_{t_i|t_{i-1}}^4}{\sigma_{t_i}^4}\|X-\hat{x}_\theta(Z_{t_i};t_i)\|^2\\&=\frac{1}{2\sigma^2_{t_{i-1}}}\frac{\alpha^2_{t_{i-1}}\sigma_{t_i|t_{i-1}}^2}{\sigma^2_{t_i}}\|X-\hat{x}_\theta(Z_{t_i};t_i)\|^2\\&=\frac{1}{2\sigma^2_{t_{i-1}}}\frac{\alpha^2_{t_{i-1}}(\sigma_{t_i}^2-\alpha^2_{t_i|t_{i-1}}\sigma^2_{t_{i-1}})}{\sigma^2_{t_i}}\|X-\hat{x}_\theta(Z_{t_i};t_i)\|^2\\&=\frac{1}{2}\frac{\alpha^2_{t_{i-1}}\sigma_{t_i}^2/\sigma^2_{t_{i-1}}-\alpha^2_{t_i}}{\sigma^2_{t_i}}\|X-\hat{x}_\theta(Z_{t_i};t_i)\|^2\\&=\frac12\left(\text{SNR}(t_{i-1})-\text{SNR}(t_i)\right)\|X-\hat{x}_\theta(Z_{t_i};t_i)\|^2.
    \end{aligned} \]
    Thus,
    \[
    \begin{aligned}\mathcal{L}(X) &=  \sum_{i=1}^T \field{E}_{q(Z_{t_i}|X)}[D(q(Z_{t_{i-1}}|Z_{t_i}, X)\| p(Z_{t_{i-1}}|Z_{t_i})]\\&=\frac12\sum_{i=1}^T\field{E}_{q(Z_{t_i}|X)}\left[\left(\text{SNR}(t_{i-1})-\text{SNR}(t_i)\right)\|X-\hat{x}_\theta(Z_{t_i};t_i)\|^2\right].
    \end{aligned}
    \]
\end{enumerate}
\end{solution}

\item {\bf Masked Diffusion} {\it (30 points)}\\
Consider a masked discrete diffusion model with absorbing matrix.
More precisely, we have $X_0\sim p_{data}$ and the forward process $X_1, X_2, \dots, X_n$ where
$X_i\sim p_i$ with $p_i = Q_\text{absorb} p_{i-1}$ for all $i$. 
Note that
\[
Q_\text{absorb} = \left[ \begin{matrix}
    1 - \epsilon & 0 & \cdots & 0 & 0 \\
    0 & 1 - \epsilon & \cdots & 0 & 0 \\
    \vdots & \vdots & \ddots & \vdots & \vdots \\
    0 & 0 & \cdots & 1 - \epsilon & 0 \\
    \epsilon & \epsilon & \cdots & \epsilon & 1
\end{matrix} \right]
\]
\begin{enumerate}
    \item Find $p(X_i=x|X_0=x)$ as a function of $i$ and $\epsilon$. {\it (10 points)}
    \item Let $Q^{(i)}$ be a transition matrix that satisfies $p_i = Q^{(i)} p_0$. Find $Q^{(i)}$. {\it (10 points)}
    \item We re-parametrize the forward process $\{Y_t\}_{t=0}^T$ so that the $Y_t$ is now has continuous time scale.
    More precisely, we have $Y_{i\epsilon} = X_i$ where $T = n\epsilon$.
    Let $q_t$ be the distribution of $Y_t$ (i.e., $q_{i\epsilon} = p_i$).
    We still have $q_{t+\epsilon} = Q_\text{absorb} q_t$.
    Find $Q_t$ that satisfies
    \begin{align*}
        \frac{1}{\epsilon} (q_{t+\epsilon} - q_t) = Q_t q_t.
    \end{align*}
     {\it (10 points)}
\end{enumerate}


\vskip .2in
\begin{solution}{\bf Solution: Masked Diffusion.}
\begin{enumerate}
\item Let \(\cX=\{1,2,\ldots,K\}\) be the state space and let \(\overline{\cX}=\cX\cup \{[\mathbf{M}]\}\), so that \(Q_\text{absorb}\in\R^{(K+1)\times (K+1)}\). Then, the transition probability at each step is given by:
\begin{equation*}
    p_{i|i-1}(y|x) = \begin{cases}
        1-\epsilon &\text{if}\quad y=x\neq[\mathbf{M}] \\
        \epsilon &\text{if}\quad x\neq y=[\mathbf{M}] \\
        0 &\text{otherwise}
    \end{cases}
\end{equation*}
Notably, transitions between two distinct unmasked tokens and from the masked token to unmasked tokens are not allowed.

By the Markov property of the diffusion forward process, we have:
\begin{equation*}
    p_{i|0}(x|x) = \sum_{x_{i-1}\in\overline{\cX}} \sum_{x_{i-2}\in\overline{\cX}} \cdots \sum_{x_1\in\overline{\cX}} p_{i|i-1}(x|x_{i-1})p_{i-1|i-2}(x_{i-1}|x_{i-2})\cdots p_{1|0}(x_1|x)
\end{equation*}

Taking into account the disallowed transitions, we obtain:
\begin{align*}
    p_{i|0}(x|x) 
    & = p_{i|i-1}(x|x)p_{i-1|i-2}(x|x)\cdots p_{1|0}(x|x) \\
    & = (1-\epsilon)(1-\epsilon)\cdots(1-\epsilon) \\
    & = (1-\epsilon)^i.
\end{align*}

\item
Observe that \(Q^{(i)}(y,x)=p_{i|0}(y|x)\).
Again, considering the disallowed transitions, for \(y \neq x\) (both unmasked), we have \(p_{i|0}(y|x) = 0\). From the previous result, \(p_{i|0}(x|x)=(1-\epsilon)^i\). Hence,
\begin{align*}
    p_{i|0}([\mathbf{M}]|x) 
    & = 1 - \sum_{y=1}^K p_{i|0}(y|x) \\
    & = 1 - p_{i|0}(x|x) \\
    & = 1 - (1-\epsilon)^i.
\end{align*}
Therefore, the matrix \(Q^{(i)}\) is:
\begin{equation*}
    Q^{(i)} = \left[ \begin{matrix}
        (1-\epsilon)^i & 0 & \cdots & 0 & 0 \\
        0 & (1-\epsilon)^i & \cdots & 0 & 0 \\
        \vdots & \vdots & \ddots & \vdots & \vdots \\
        0 & 0 & \cdots & (1-\epsilon)^i & 0 \\
        1-(1-\epsilon)^i & 1-(1-\epsilon)^i & \cdots & 1-(1-\epsilon)^i & 1
    \end{matrix} \right].
\end{equation*}

\paragraph{Alternative approach.}  
Alternatively, \(Q^{(i)}\) can be derived by computing the \(i\)-th power of the matrix \(Q_\text{absorb}\). This power can be computed either by induction or by using the decomposition \(Q_\text{absorb}=(1-\epsilon)I+\epsilon A\) together with the binomial expansion.

\item
\begin{align*}
    q_{t+\epsilon}-q_t
    & = Q_\text{absorb}q_t - q_t \\
    & = (Q_\text{absorb}-I)q_t, \\
\end{align*}
where \(I\) is the identity matrix. Thus, 
\begin{align*}
    Q_t 
    & = \frac{1}{\epsilon}(Q_\text{absorb}-I) \\
    & = \frac{1}{\epsilon}
    \begin{bmatrix}
    - \epsilon & 0 & \cdots & 0 & 0 \\
    0 & - \epsilon & \cdots & 0 & 0 \\
    \vdots & \vdots & \ddots & \vdots & \vdots \\
    0 & 0 & \cdots & - \epsilon & 0 \\
    \epsilon & \epsilon & \cdots & \epsilon & 0
    \end{bmatrix}
    = \left[ \begin{matrix}
    - 1 & 0 & \cdots & 0 & 0 \\
    0 & - 1 & \cdots & 0 & 0 \\
    \vdots & \vdots & \ddots & \vdots & \vdots \\
    0 & 0 & \cdots & - 1 & 0 \\
    1 & 1 & \cdots & 1 & 0
    \end{matrix} \right].
\end{align*}    
\end{enumerate}
\end{solution}

\item {\bf Discrete Langevin} {\it (40 points)}\\
In the homework problem, we showed that the Langevin SDE 
\[
dX_t=\frac{1}{2}\nabla_{X_t} \log p(X_t)dt+dW_t.
\]
preserves the distribution $p$ for all $t$, if the initial distribution satisfies $p_0=p$.
However, this result can be significantly strengthened. Specifically, even when the initial distribution $p_0$ differs from $p$,
the distribution of $X_t\sim p_t$ will converge to the target distribution $p$ as $t\rightarrow \infty$.

In this problem, we examine this convergence behavior in a simple setting where the target distribution $p$ is Gaussian.
Consider the probability density function of the Gaussian distribution $\mathcal{N}(\mu, \sigma^2)$:
\begin{align*}
    p(x) = \frac{1}{\sqrt{2\pi \sigma^2}} \exp\left(-\frac{(x-\mu)^2}{2\sigma^2}\right).
\end{align*}
Let this Gaussian $p(x)$ be our target distribution.

\begin{enumerate}
    \item Compute the score function $\frac{\partial}{\partial x} \log p(x)$. {\it (10 points)}
    \item Consider the discretized Langevin SDE with a small $\epsilon$:
    \begin{align*}
        X_{i+1} = X_i + \epsilon \cdot \frac{\partial}{\partial x}\log p(X_i) +\sqrt{2\epsilon}Z_i
    \end{align*}
    where $Z_i\sim \mathcal{N}(0, 1)$ is an independent Gaussian noise.
    Suppose the initial distribution is also standard Gaussian $X_0 \sim \mathcal{N}(0, 1)$,
    which differes from the target distribution $p$.
    Show that $X_i$ is Gaussian for all $i$. {\it (10 points)}
    \item Our goal is to show that $p_{X_n}$ converges to $p$ as $n$ grows.
    Since $X_i$ is Gaussian, we can write $X_i\sim \mathcal{N}(\mu_i, \sigma_i^2)$.
    Derive the recursive relations for $\mu_{i+1}$ in terms of $\mu_i$, and $\sigma_{i+1}^2$ in terms of $\sigma_i^2$.
    Show that $\mu_n\rightarrow \mu$ as $n\rightarrow \infty$. {\it (10 points)}
    \item Using a suitable approximation, show that $\sigma_n^2\rightarrow \sigma^2$ as $n\rightarrow \infty$,
    assuming $\epsilon$ is small.
     {\it (10 points)}
\end{enumerate}

\vskip .2in
\begin{solution}{\bf Solution: Discrete Langevin.}
\begin{enumerate}
\item The score function is simply
\begin{align*}
    \frac{\partial}{\partial x} \log p(x) = -\frac{x-\mu}{\sigma^2}.
\end{align*}
\item Using the above form of the score function, the discretized Langevin SDE is
    \begin{align*}
        X_{i+1} =& X_i - \epsilon \cdot \frac{X_i-\mu}{\sigma^2} +\sqrt{2\epsilon}Z_i\\
        =& \left(1-\frac{\epsilon}{\sigma^2}\right) X_i + \frac{\epsilon}{\sigma^2}\mu +\sqrt{2\epsilon}Z_i
    \end{align*}
    Since $\epsilon\mu/\sigma^2$ is a constant, $X_{i+1}$ is a linear combination of $X_i$ and $Z_i$.
    Thus, if $X_i$ is Gaussian, then $X_{i+1}$ is also Gaussian.
    
\item Using the above recusive relation, we have
\begin{align*}
    \mu_{i+1} = \left(1-\frac{\epsilon}{\sigma^2}\right) \mu_i + \frac{\epsilon}{\sigma^2}\mu
\end{align*}
which is a convex combination of $\mu_i$ and $\mu$.
Thus, if we iteratively apply this relation, we have $\mu_i \rightarrow \mu$.
Or simply, we can take the limit to both sides, we have
\begin{align*}
    \lim \mu_{n} = \left(1-\frac{\epsilon}{\sigma^2}\right) \lim \mu_n + \frac{\epsilon}{\sigma^2}\mu
\end{align*}
which implies $\lim \mu_n = \mu$.

Similarly, we have
\begin{align*}
    \sigma_{i+1}^2 = \left(1-\frac{\epsilon}{\sigma^2}\right)^2 \sigma_i^2 + 2\epsilon.
\end{align*}

\item If we take the limit to both sides, we have
\begin{align*}
    \lim \sigma_{n}^2 = \left(1-\frac{\epsilon}{\sigma^2}\right)^2 \lim \sigma_n^2 + 2\epsilon.
\end{align*}
which implies
\begin{align*}
    \lim \sigma_{n}^2 =& \frac{2\epsilon}{1-(1-\epsilon/\sigma^2)^2}\\
    \approx& \frac{2\epsilon}{1 - (1-2\epsilon/\sigma^2)}\\
    =& \sigma^2.
\end{align*}
Note that the approximation is valid when $\epsilon$ is small.
\end{enumerate}
\end{solution}
