\item \textbf{Approximate DP:} Consider the following mechanisms $M$ that takes a dataset $x\in [0,1]^n$ and returns
an estimate of the mean $\bar{x} = (\sum_{i=1}^n x_i)/n$.

\begin{enumerate}[(M1)]
    \item $M_1(x) = \bar{x}+[Z]^1_{-1}$, for $Z\sim \mathrm{Lap}(2/n)$.
    \item 
    $$M_2(x) = \begin{cases} 1 & \text{w.p. } \overline{x}\\
    0 & \text{w.p. } 1-\overline{x}.
    \end{cases}.$$
    
\end{enumerate}

In HW3, we have identified the above mechanisms that do not meet the definition of $(\epsilon,0)$-differential privacy.
For those mechanisms, calculate the smallest value of $\delta$ (again possibly as a function of $n$) 
for which they satisfy $(\epsilon,\delta)$ differential privacy for a finite value of $\epsilon$.

\begin{solution}
\textbf{Solution.}\\
\begin{enumerate}
    \item 
    % First of all, we have
    % \begin{align*}
    %     \delta =& Pr(1-1/n \leq [Z]_{-1}^1 \leq 1)\\
    %     =& Pr(1-1/n\leq Z)\\
    %     =& \frac{1}{2} e^{-(n/2)(1-1/n)}\\
    %     =&\frac{1}{2} e^{-n+1/2}.
    % \end{align*}
    % Similar to the regular Laplacian analysis, $\epsilon$ should be $1/2$.
    Suppose $\bar{x} = 1/n$ and $\bar{x'}=0$.
    For any $a$, we have 
    \begin{align*}
        Pr(a\leq 1/n+[Z]_{-1}^1\leq 1+1/n) \leq e^\epsilon Pr(a\leq [Z]_{-1}^1\leq 1+1/n) + \delta.
    \end{align*}
    If $a>1$, we have
    \begin{align*}
        Pr(a-1/n\leq [Z]_{-1}^1\leq 1) \leq \delta
    \end{align*}
    which implies
    \begin{align*}
        \delta =& Pr(1-1/n \leq [Z]_{-1}^1 \leq 1)\\
        =& Pr(1-1/n\leq Z)\\
        =& \frac{1}{2} e^{-(n/2)(1-1/n)}\\
        =&\frac{1}{2} e^{-n/2+1/2}.
    \end{align*}

    On the other hand, if $a<1$, we have
    \begin{align*}
        &Pr(a-1/n\leq [Z]_{-1}^1\leq 1) \leq e^\epsilon Pr(a\leq [Z]_{-1}^1\leq 1+1/n) + \delta\\
        \Leftrightarrow &  Pr(a-1/n\leq [Z]_{-1}^1\leq 1-1/n) \leq e^\epsilon Pr(a\leq [Z]_{-1}^1\leq 1) 
    \end{align*}
    which is true for $\epsilon = 1/2$.
    Thus, the above mechanism is $(1/2, 1/2 e^{-n+1/2})$-DP.

    Note that you should specify $\epsilon$ as well since the mechanism is not $(\epsilon, \delta)$-DP for $\epsilon<1/2$.

    \item 
    \begin{align*}
        &Pr(M_2(x) = 1) \leq e^\epsilon Pr(M_2(x')=1)+\delta
        \Leftrightarrow \bar{x} \leq e^\epsilon \bar{x'} + \delta.
    \end{align*}
    Thus, the $\delta$ should be
    \begin{align*}
        \delta = \max_{x, x'} \bar{x} - e^\epsilon \bar{x'} = 1/n.
    \end{align*}
    Thus, the above mechanism is $(0, 1/n)$-DP. 
    Note that you should clearly mention that the mechanism is $(\epsilon, \delta)$-DP for all $\epsilon$.
    \end{enumerate}
\end{solution}

\item \textbf{Regression: } 
Consider a dataset where each of its $n$ rows is a pair of real numbers $(x_i,y_i)$, each from an interval $[-b,b]$.
Suppose we wish to find a best-fit linear relationship $y_i \approx \beta x_i$ between the $y$'s and the $x$'s.
Non-privately, a standard way to estimate $\beta$ is via the OLS regression formula 
$$\hat{\beta} = \hat{\beta}(x,y) =  \frac{S_{xy}}{S_{xx}}
= \frac{\sum_i x_iy_i}{\sum_i x_i^2}.$$
This is called {\em ordinary least-squares (OLS)} regression, since $\hat{\beta}$ is the minimizer of the mean-squared residuals 
\begin{equation} \label{eqn:residuals}
    \frac{1}{n} \sum_i (y_i -\hat{\beta} x_i)^2.
\end{equation}

\begin{enumerate}
    \item Show that the function $\hat{\beta}(x,y)$ has infinite global sensitivity, and hence we cannot get a useful DP estimate of it via a direct application of the Laplace or Gaussian mechanisms.
    \item Show that $S_{xy}$ and $S_{xx}$ have global sensitivity that is bounded solely as a function of $b$, and hence each of these can be approximated in a DP manner using the Laplace mechanism. \label{part:suffstats}
    \item Using Part~\ref{part:suffstats} together with basic composition and post-processing,
    devise and implement an $\epsilon$-DP algorithm for approximating $\hat{\beta}$ on an arbitrary dataset with $x_i,y_i\in \fR$.
    In addition to the dataset $((x_1,y_1),\ldots,(x_n,y_n))$,
    your implementation should take as input parameters a clipping bound $b$ and the privacy-loss parameter $\epsilon$.

    

\end{enumerate}

\begin{solution}
\vskip .2in
\textbf{Solution.}\\
\begin{enumerate}
    \item Let $X = \{(1, 0), \dots, (0, 0)\}$ and $X' = \{(u, v), (0, 0), \dots, (0, 0)\}$.
    Then, $\hat{\beta}(X) = 0$, and $\hat{\beta}(X') = v/u$.
    Since $u$ can be arbitrarily close to 0 and the difference $v/u$ is unbounded.
    \item Sensitivity of $S_{xy}$ is $b\cdot b - b \cdot (-b) = 2b^2$ while sensitivity of $S_{xx}$ is $b\cdot b - 0 \cdot 0 = b^2$.
    \item Based on composition theorem, you can mix two privatization with $\alpha \epsilon$ and $(1-\alpha)\epsilon$.
        In other words, compute $\tilde{S}_{xx}(X) = S_{xx}(X) + Z_{xx}$ where $Z_{xx}$ be Laplace random variable with parameter $b^2/\alpha\epsilon$.
        Then, compute $\tilde{S}_{xy}(X) = S_{xy}(X) + Z_{xy}$ where $Z_{xy}$ be a Laplace random variable with parameter $2b^2/(1-\alpha)$.
        Then, we compute $\tilde{\beta} = \tilde{S}_{xy}/ \tilde{S}_{xx}$.
        The combining $\tilde{S}_{xx}$ and $\tilde{S}_{xy}$ is $\epsilon$-DP due to composition theorem,
        and the final division does not affect the overall privacy due to post processing property.

        Finally, you need to discuss the best $\alpha$ that minimizes the (any reasonable) loss between $\beta$ and $\hat{\beta}$.

        
\end{enumerate}
\end{solution}

\item 
\begin{enumerate}
    \item  Show that $\sigma(x) = \sin (x)$ is discriminatory. 
    \item  Show that $\sigma(x) = \sin (x) + \cos(x)$ is discriminatory. 
\end{enumerate}

\begin{solution}
\vskip .2in
\textbf{Solution.}\\
\begin{enumerate}
    \item 
Let $\mu$ be a measure, then
\begin{align}
    \int \sin (a^\intercal x) d\mu(x) = 0
\end{align}
for all $a$.
Also,
\begin{align}
    \int \sin(-a^\intercal x+\pi/2) d\mu(x) =&
    \int \cos(a^\intercal x) d\mu(x)\\ =& 0.
\end{align}
Thus, the Fourier transform of $\mu$ is 
\begin{align}
    \hat{\mu}(a) = \int e^{ia^\intercal x} d\mu(x) = 0
\end{align}
for all $a$, which implies $\mu\equiv 0$.
Thus, $\sin(x)$ is discriminatory.

\item 
Let $\mu$ be a measure, then
\begin{align}
    \int \sin (a^\intercal x) + \cos(a^\intercal x)d\mu(x) = 0
\end{align}
for all $a$.
Also,
\begin{align}
    \int \sin(-a^\intercal x) + \cos(-a^\intercal x) d\mu(x) =&
    \int -\sin(a^\intercal x) + \cos(a^\intercal x) d\mu(x)\\ =& 0.
\end{align}
Thus, we have
\begin{align}
    \int \sin (a^\intercal x) d\mu(x) = \int \cos (a^\intercal x) d\mu(x) = 0
\end{align}
Finally, 
\begin{align}
    \hat{\mu}(a) = \int e^{ia^\intercal x} d\mu(x) = 0
\end{align}
for all $a$, which implies $\mu\equiv 0$.
Thus, $\sin(x)$ is discriminatory.
\end{enumerate}

\end{solution}

\item Let $\Omega\subseteq\fR^d$ be compact and $\sigma\in \cC(\fR)$ is discriminatory function.
Let 
\[
\cN^{(1)}(\sigma,d,N)=\{A\cdot +b\,|\,A\in \fR^{N\times d},\,b\in \fR^{N}\}
\]
and
\[
\cN^{(k+1)}(\sigma,d,N)=\{A\sigma(f(\cdot))+b\,|\,
N_\mathrm{prev}\in \mathbb{N},\,
A\in \fR^{N\times N_\mathrm{prev}},\,b\in \fR^{N},\,f\in \cN^{(k)}(\sigma,d,N_\mathrm{prev})\},
\]
where $\sigma$ applies elementwise, for $k=1,2,\dots$.
\begin{enumerate}
    \item Briefly describe why $\cN^{(2)}(\sigma, d, 1)$ is dense in $(C(\Omega), \|\cdot \|_\infty)$.\\
    (Hint: We did this in our lecture).
    \item Show that $\cN^{(2)}(\sigma, d, m)$ is dense in  $(C(\Omega;\fR^m), \|\cdot \|_\infty)$ for all $m\in\fN$
    where $C(\Omega; \fR^m)$ is the space of bounded continuous function $u:\Omega\rightarrow \fR^m$.
    \item Show that $\cN^{(L)}(\sigma,d,1)$ is dense in $(\cC(\Omega),\|\cdot\|_\infty)$ for $L\in \{2,3,\ldots\}$. 
\end{enumerate}
\begin{solution}
\vskip .2in
\textbf{Solution.}\\
\begin{enumerate}
    \item $\cN^{(2)}(\sigma, d, 1) = \mbox{span}(\{\sigma(a^\intercal x+b):a\in \fR^{1\times d}, b\in\fR\}$, which we showed dense in $(C(\Omega), \|\cdot\|_\infty)$.

    \item Let $g = (g_1, \dots, g_m)\in C(\Omega, \fR^m)$.
    For any $\epsilon > 0$, there exists $h_1, \dots, h_m\in \cN^{(2)}(\sigma, d, 1)$ such that $\|g_i-h_i\|_\infty < \epsilon/m$.
    Suppose $h_i$ has width $N_i$, then we can construct the 2-layer network $h$ with width $\sum_{i=1}^m N_i$ where
    \begin{align*}
        h(x) = (h_1(x), h_2(x), \dots, h_m(x)).
    \end{align*}
    It is clearly $h\in\cN^{(2)}(\sigma, d, m)$ and 
    \begin{align}
        \|h-g\|_\infty \leq \sum_{i=1}^m \|h_i-g_i\|_\infty < \epsilon.
    \end{align}
    Thus, $\cN^{(2)}(\sigma, d, m)$ is dense in  $(C(\Omega;\fR^m), \|\cdot \|_\infty)$.

    \item We will show that $\cN^{(L)}(\sigma, d, d)$ is dense for all $L\geq 2$ using induction.
    For $L\geq 3$, suppose $\cN^{(L-1)}(\sigma, d, d)$ is dense.
    Let $g\in (C(\Omega; \fR^d, \|\cdot\|_\infty)$ and $\epsilon> 0$.
    First, since $\cN^{(2)}(\sigma, d, d)$ is dense, there exists $h_2=A_2\sigma(A_1x+b_1)+b_1\in\cN^{(2)}(\sigma, d, d)$ such that
    \begin{align*}
        \|h_2-g\|_\infty < \epsilon/2.
    \end{align*}
    On the other hand, since $\cN^{(L-1)}(\sigma, d, d)$ is dense, there exists $\tilde{h}\in\cN^{(L-1)}(\sigma, d, d)$ such that
    \begin{align}
        \|\tilde{h} - (A_1\cdot +b_1)\|_\infty < \epsilon'
    \end{align}
    where $\epsilon'>0$ will be specified later.

    Let $h=A_2\sigma(\tilde{h}(x))+b_2\in\cN^{(L)}(\sigma, d, d)$.
    Then,
    \begin{align*}
        \|h(x) - h_2(x)\| =& \|A_2\sigma(\tilde{h}(x)) + b_1 - (A_2\sigma(A_1x+b)+b_2)\|\\
        =& \|A_2(\sigma(\tilde{h}(x)) -\sigma(A_1x+b))\|
    \end{align*}
    Since $\sigma$ is continuous, we can set $\epsilon'$ small enough so that $\|h - h_2\|_\infty < \epsilon/2$.
    Finally,
    \begin{align}
        \|h-g\|_\infty \leq \|h-h_2\|_\infty + \|h_2-g\|_\infty < \epsilon.
    \end{align}
    
    
\end{enumerate}
\end{solution}
 
\end{enumerate}
