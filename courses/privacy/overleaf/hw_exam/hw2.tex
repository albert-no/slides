\item \textbf{Randomized Response, re-revisited:}
We will see some generalizations of randomized response, beyond just binary alphabets.
I will informally and vaguely describe an algorithm, you must rigorously define and specify the algorithm and prove that it is $\epsilon$-differentially
private.
\begin{enumerate}
\item Assume $X_i \in \{1, \dots, k\}$ for the remainder of this problem.
The vector $(Y_1, \dots , Y_n)\in \{1, \dots, k\}^n$ is output, where $Y_i$ is equal to $X_i$ with probability proportional to $g(\epsilon)$ 
(for some function $g$ which you must specify), and equal to each $s \in \{1, . . . , k\} \setminus X_i$ with probability proportional to 1.
Specify the algorithm rigorously (define probability of randomized response) and prove that it is $\epsilon$-DP.

\item Here is another way to generalize randomized response. 
The vector $(Y_1, \dots , Y_n) \in \{0, 1\}^{kn}$ is output. $Y_i \in \{0, 1\}^k$ is a vector generated in the following manner:
each $X_i$ is first converted to a ``one-hot'' vector $\in \{0, 1\}^k$, where coordinate $j$ is 1 if $j = X_i$ and 0 otherwise.
$Y_i$ generated from $X_i$ by applying a bitwise randomized response (with appropriate parameter).
Specify the algorithm rigorously (define probability of randomized response) and prove that it is $\epsilon$-DP.
\end{enumerate}

\begin{solution}
{\bf Solution: }\\
\begin{enumerate}
    \item We can set $g(\epsilon) = e^{\epsilon}$, so that
    \begin{align*}
        Pr(Y_i=X_i) =& \frac{e^\epsilon}{e^\epsilon +(k-1)}\\
        Pr(Y_i=s) =& \frac{1}{e^\epsilon +(k-1)}
    \end{align*}
    for $s\neq X_i$
    The probability ratio is always bounded by $e^{\epsilon}$.

    \item Without loss of generality, 
    consider $X$ and $X'$ where $X_1 = (1, 0, \dots, 0)$ and $X_1'= (0, 1, 0, \dots, 0)$ (and all others are the same $X_i=X'_i$).
    \begin{align*}
        \frac{Pr(Y_1= (b_1, \dots, b_k))}{Pr(Y'_1=(b_1, \dots, b_k))} 
        =& \frac{Pr(Y_{11}, Y_{12} = b_1, b_2)}{Pr(Y'_{11}, Y'_{12}=b_1, b_2)}.
    \end{align*}
    When the bit-flipping probability is $q < 1/2$, the maximum ratio would be $(1-q)^2/q^2$ which should be bounded by $\epsilon$.
    Thus, the flipping probability should be
    \begin{align*}
        q = \frac{e^{\epsilon/2}}{1+e^{\epsilon/2}}.
    \end{align*}
\end{enumerate}
\end{solution}

\item \textbf{Approximate DP:} Consider the following mechanisms $M$ that takes a dataset $x\in [0,1]^n$ and returns
an estimate of the mean $\bar{x} = (\sum_{i=1}^n x_i)/n$.

\begin{enumerate}[(M1)]
    \item $M_1(x) = \bar{x}+[Z]^1_{-1}$, for $Z\sim \mathrm{Lap}(2/n)$.
    \item 
    $$M_2(x) = \begin{cases} 1 & \text{w.p. } \overline{x}\\
    0 & \text{w.p. } 1-\overline{x}.
    \end{cases}.$$
    
\end{enumerate}
The above mechanisms do not meet the definition of $(\epsilon,0)$-differential privacy.
For those mechanisms, calculate the smallest value of $\delta$ (again possibly as a function of $n$) 
for which they satisfy $(\epsilon,\delta)$ differential privacy for a finite value of $\epsilon$.

\begin{solution}
\textbf{Solution.}\\
\begin{enumerate}
    \item 
    % First of all, we have
    % \begin{align*}
    %     \delta =& Pr(1-1/n \leq [Z]_{-1}^1 \leq 1)\\
    %     =& Pr(1-1/n\leq Z)\\
    %     =& \frac{1}{2} e^{-(n/2)(1-1/n)}\\
    %     =&\frac{1}{2} e^{-n+1/2}.
    % \end{align*}
    % Similar to the regular Laplacian analysis, $\epsilon$ should be $1/2$.
    Suppose $\bar{x} = 1/n$ and $\bar{x'}=0$.
    For any $a$, we have 
    \begin{align*}
        Pr(a\leq 1/n+[Z]_{-1}^1\leq 1+1/n) \leq e^\epsilon Pr(a\leq [Z]_{-1}^1\leq 1+1/n) + \delta.
    \end{align*}
    If $a>1$, we have
    \begin{align*}
        Pr(a-1/n\leq [Z]_{-1}^1\leq 1) \leq \delta
    \end{align*}
    which implies
    \begin{align*}
        \delta =& Pr(1-1/n \leq [Z]_{-1}^1 \leq 1)\\
        =& Pr(1-1/n\leq Z)\\
        =& \frac{1}{2} e^{-(n/2)(1-1/n)}\\
        =&\frac{1}{2} e^{-n/2+1/2}.
    \end{align*}

    On the other hand, if $a<1$, we have
    \begin{align*}
        &Pr(a-1/n\leq [Z]_{-1}^1\leq 1) \leq e^\epsilon Pr(a\leq [Z]_{-1}^1\leq 1+1/n) + \delta\\
        \Leftrightarrow &  Pr(a-1/n\leq [Z]_{-1}^1\leq 1-1/n) \leq e^\epsilon Pr(a\leq [Z]_{-1}^1\leq 1) 
    \end{align*}
    which is true for $\epsilon = 1/2$.
    Thus, the above mechanism is $(1/2, 1/2 e^{-n+1/2})$-DP.

    Note that you should specify $\epsilon$ as well since the mechanism is not $(\epsilon, \delta)$-DP for $\epsilon<1/2$.

    \item 
    \begin{align*}
        &Pr(M_2(x) = 1) \leq e^\epsilon Pr(M_2(x')=1)+\delta
        \Leftrightarrow \bar{x} \leq e^\epsilon \bar{x'} + \delta.
    \end{align*}
    Thus, the $\delta$ should be
    \begin{align*}
        \delta = \max_{x, x'} \bar{x} - e^\epsilon \bar{x'} = 1/n.
    \end{align*}
    Thus, the above mechanism is $(0, 1/n)$-DP. 
    Note that you should clearly mention that the mechanism is $(\epsilon, \delta)$-DP for all $\epsilon$.
    \end{enumerate}
\end{solution}

\item \textbf{Regression: } 
Consider a dataset where each of its $n$ rows is a pair of real numbers $(x_i,y_i)$, each from an interval $[-b,b]$.
Suppose we wish to find a best-fit linear relationship $y_i \approx \beta x_i$ between the $y$'s and the $x$'s.
Non-privately, a standard way to estimate $\beta$ is via the OLS regression formula 
$$\hat{\beta} = \hat{\beta}(x,y) =  \frac{S_{xy}}{S_{xx}}
= \frac{\sum_i x_iy_i}{\sum_i x_i^2}.$$
This is called {\em ordinary least-squares (OLS)} regression, since $\hat{\beta}$ is the minimizer of the mean-squared residuals 
\begin{align*}
    \frac{1}{n} \sum_i (y_i -\hat{\beta} x_i)^2.
\end{align*}

\begin{enumerate}
    \item Show that the function $\hat{\beta}(x,y)$ has infinite global sensitivity, and hence we cannot get a useful DP estimate of it via a direct application of the Laplace or Gaussian mechanisms.
    \item Show that $S_{xy}$ and $S_{xx}$ have global sensitivity that is bounded solely as a function of $b$, and hence each of these can be approximated in a DP manner using the Laplace mechanism. \label{part:suffstats}
    \item Using Part~\ref{part:suffstats} with basic composition and post-processing,
    devise and implement an $\epsilon$-DP algorithm for approximating $\hat{\beta}$ on an arbitrary dataset with $x_i,y_i\in \fR$.
    In addition to the dataset $((x_1,y_1),\ldots,(x_n,y_n))$,
    your implementation should take as input parameters a clipping bound $b$ and the privacy-loss parameter $\epsilon$.

\end{enumerate}

\begin{solution}
\vskip .2in
\textbf{Solution.}\\
\begin{enumerate}
    \item Let $X = \{(1, 0), \dots, (0, 0)\}$ and $X' = \{(u, v), (0, 0), \dots, (0, 0)\}$.
    Then, $\hat{\beta}(X) = 0$, and $\hat{\beta}(X') = v/u$.
    Since $u$ can be arbitrarily close to 0 and the difference $v/u$ is unbounded.
    \item Sensitivity of $S_{xy}$ is $b\cdot b - b \cdot (-b) = 2b^2$ while sensitivity of $S_{xx}$ is $b\cdot b - 0 \cdot 0 = b^2$.
    \item Based on composition theorem, you can mix two privatization with $\alpha \epsilon$ and $(1-\alpha)\epsilon$.
        In other words, compute $\tilde{S}_{xx}(X) = S_{xx}(X) + Z_{xx}$ where $Z_{xx}$ be Laplace random variable with parameter $b^2/\alpha\epsilon$.
        Then, compute $\tilde{S}_{xy}(X) = S_{xy}(X) + Z_{xy}$ where $Z_{xy}$ be a Laplace random variable with parameter $2b^2/(1-\alpha)$.
        Then, we compute $\tilde{\beta} = \tilde{S}_{xy}/ \tilde{S}_{xx}$.
        The combining $\tilde{S}_{xx}$ and $\tilde{S}_{xy}$ is $\epsilon$-DP due to composition theorem,
        and the final division does not affect the overall privacy due to post processing property.

        Finally, you need to discuss the best $\alpha$ that minimizes the (any reasonable) loss between $\beta$ and $\hat{\beta}$.

        
\end{enumerate}
\end{solution}

\item For $f : \mathcal{X}^n \to \mathbb{R}$ with $\ell_1$ sensitivity $\Delta$, the truncated Laplacian mechanism is
\[
\mathcal{M}(X) = f(X) + Z
\]
where $Z$ is $\mathrm{LapTr}(\lambda, \tau)$. Recall that the Truncated Laplacian random variable has pdf
\[
f_Z(z) =
\begin{cases}
C \exp(-|z|/\lambda) & \text{if } |z| \leq \tau \\
0 & \text{if } |z| > \tau
\end{cases}
\]
where $C$ is a normalization constant.

\begin{enumerate}
    \item[(a)] Find $C$.
    \item[(b)] Show that the Truncated Laplacian mechanism is $(\varepsilon, \delta)$-DP.
    You may assume $\lambda = \Delta / \varepsilon$ and $\tau < \Delta$, but you need to specify $\delta$.
\end{enumerate}

%%%%%%%%%%%%%%%%%%%%%%%%%% Problem 1 Solution %%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{solution}
\vskip .2in
\textbf{Solution.}\\
\begin{enumerate}
\item[(a)] Note that
\begin{align*}
    \int_{-\tau}^{\tau} \exp(-|z|/\lambda) \, dz 
    & = 2 \int_0^{\tau} \exp(-z/\lambda) \, dz \\
    & = 2\lambda (1 - \exp(-\tau/\lambda)).
\end{align*}
Thus,
\[
C = \frac{1}{2\lambda (1 - \exp(-\tau/\lambda))}.
\]

\item[(b)] Let $\lambda = \Delta/\varepsilon$. For $y$ such that $|f(X) - y| \leq \tau$ and $|f(X') - y| \leq \tau$, the privacy loss random variable
\begin{align*}
    I_{X,X'}(y) 
    & = \log \frac{\Pr[\mathcal{M}(X) = y]}{\Pr[\mathcal{M}(X') = y]} \\
    & = -\frac{|f(X) - y|}{\lambda} + \frac{|f(X') - y|}{\lambda} \\
    & \leq \frac{|f(X) - f(X')|}{\lambda} \\
    & \leq \frac{\Delta}{\lambda}\\
    & = \varepsilon.
\end{align*}

Now, consider the probability of bad events
\begin{align*}
\Pr[|f(X) - y| > \tau \text{ or } |f(X') - y| > \tau]
&= \Pr[|Z| > \tau \text{ or } |f(X') - f(X) - Z| > \tau] \\
&\leq \Pr[|\Delta - Z| > \tau] \\
&= \Pr[Z > \tau + \Delta \text{ or } Z < \Delta - \tau] \\
&= \int_{-\tau}^{\Delta - \tau} C \exp(-|z|/\lambda) \, dz \\
&= -C\lambda \left( \exp(-\tau/\lambda) - \exp(-(\tau - \Delta)/\lambda) \right) \\
&= -C\lambda \exp(-\tau/\lambda)(1 - \exp(\Delta/\lambda)) \\
&= \frac{-\exp(-\tau/\lambda)(1 - \exp(\Delta/\lambda))}{2(1 - \exp(-\tau/\lambda))}.
\end{align*}

Thus, it is $(\varepsilon, \delta)$-DP when
\[
\delta = \frac{\exp(-\tau \varepsilon / \Delta)(\exp(\varepsilon) - 1)}{2(1 - \exp(-\tau \varepsilon / \Delta))}.
\]
\end{enumerate}

\end{solution}

\begin{solution}

\textbf{Solution.}\\
\begin{enumerate}
    \item[(a)] Consider $X = ((1,0), (0,1), (0,1), \dots, (0,1))$ and $X' = ((0,1), (0,1), (0,1), \dots, (0,1))$.\\
    Probability of $\Pr[M_S(X) = (1,0)] = 1/n$ but $\Pr[M_S(X') = (1,0)] = 0$.
    
    \item[(b)] It is $(0, 1/n)$-DP.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%% Problem 3 %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\vskip .2in
\item Suppose $M : \{0,1\}^n \rightarrow \mathcal{Y}$ is an $(\varepsilon, \delta)$-differentially private mechanism, and let $A : \mathcal{Y} \rightarrow \{0,1\}^n$ be an adversary attempting to reconstruct the sensitive bits in a dataset $X \in \{0,1\}^n$ from the output $M(X)$. More precisely, given $M(X) = Y$, the adversary generates $A(Y) = \hat{X} = (\hat{X}_1, \dots, \hat{X}_n)$, and wants to maximize the fraction of correct guesses
\[
\frac{1}{n} \sum_{i=1}^n \mathbf{1}(X_i = \hat{X}_i)
\]
where
\[
\mathbf{1}(X_i = \hat{X}_i) =
\begin{cases}
1 & \text{if } X_i = \hat{X}_i \\
0 & \text{if } X_i \ne \hat{X}_i.
\end{cases}
\]

Assume the dataset is a random variable $X = (X_1, \dots, X_n)$ consisting of $n$ independent draws from a Bernoulli($p$) distribution, for a known value of $p < 1/2$.

\begin{enumerate}
    \item[(a)] Let $Z_i = \mathbf{1}(X_i = \hat{X}_i)$. Show that
    \[
    \mathbb{E}[Z_i] = \Pr[\hat{X}_i = 1 \mid X_i = 1]p + \Pr[\hat{X}_i = 0 \mid X_i = 0](1 - p)
    \]

    \item[(b)] Since $M$ is $(\varepsilon, \delta)$-DP, for any neighboring dataset $X \sim X'$, we have
    \[
    \Pr[\hat{X}_i = 1 \mid X] \leq e^{\varepsilon} \Pr[\hat{X}_i = 1 \mid X'] + \delta
    \]
    Then, show that
    \[
    \Pr[\hat{X}_i = 1 \mid X_i = 1] \leq e^{\varepsilon} \Pr[\hat{X}_i = 1 \mid X_i = 0] + \delta
    \]

    \item[(c)] Prove that the expected fraction of bits that the adversary successfully reconstructs is not much larger than the trivial bound of $1 - p$, which could be achieved by always guessing the all-zero dataset. Specifically, show that:
    \[
    \mathbb{E} \left[ \frac{1}{n} \sum_{i=1}^n Z_i \right] \leq e^{\varepsilon} \cdot (1 - p) + \delta.
    \]
    
\end{enumerate}


%%%%%%%%%%%%%%%%%%%%%%%%%% Problem 3 Solution %%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{solution}
\vskip .2in
\textbf{Solution.}\\
\begin{enumerate}
\item[(a)] We have
\begin{align*}
\mathbb{E}[Z_i] &= \Pr[Z_i = 1] \\
&= \Pr[\hat{X}_i = X_i] \\
&= \Pr[\hat{X}_i = X_i \mid X_i = 1] \Pr[X_i = 1] + \Pr[\hat{X}_i = X_i \mid X_i = 0] \Pr[X_i = 0] \\
&= \Pr[\hat{X}_i = 1 \mid X_i = 1]p + \Pr[\hat{X}_i = 0 \mid X_i = 0](1 - p).
\end{align*}

\item[(b)] 
\begin{align*}
& \Pr[\hat{X}_i = 1 \mid X_i = 1] \\
&= \sum_{X : X_i = 1} \Pr[X = (X_1, \dots, X_{i-1}, 1, X_{i+1}, \dots, X_n) \mid X_i = 1] \Pr[\hat{X}_i = 1 \mid X] \\
&\leq \sum_{X : X_i = 1} \Pr[X = (X_1, \dots, X_{i-1}, 0, X_{i+1}, \dots, X_n) \mid X_i = 0] \Pr[\hat{X}_i = 1 \mid X] \\
&\leq \sum_{X' : X_i' = 0} \Pr[X' = (X_1', \dots, 0, \dots, X_n')] (e^{\varepsilon} \Pr[\hat{X}_i = 1 \mid X'] + \delta) \\
&= e^{\varepsilon} \Pr[\hat{X}_i = 1 \mid X_i = 0] + \delta.
\end{align*}

\item[(c)] Similar from part (b), we have
\[
\Pr[\hat{X}_i = 0 \mid X_i = 0] \leq e^{\varepsilon}(1 - \Pr[\hat{X}_i = 1 \mid X_i = 1]) + \delta.
\]
Thus,
\begin{align*}
\mathbb{E}[Z_i] 
&= \Pr[\hat{X}_i = 1 \mid X_i = 1]p + \Pr[\hat{X}_i = 0 \mid X_i = 0](1 - p) \\
&\leq \Pr[\hat{X}_i = 1 \mid X_i = 1]p + (1 - p)(e^{\varepsilon} - e^{\varepsilon} \Pr[\hat{X}_i = 1 \mid X_i = 1]) + (1 - p)\delta \\
&= (1 - p)e^{\varepsilon} + (p - e^{\varepsilon}(1 - p)) \Pr[\hat{X}_i = 1 \mid X_i = 1] + (1 - p)\delta \\
&\leq (1 - p)e^{\varepsilon} + \delta.
\end{align*}

Then, the average is clearly bounded by the same:
\[
\mathbb{E}\left[\frac{1}{n} \sum_{i=1}^n Z_i\right] = \frac{1}{n} \sum_{i=1}^n \mathbb{E}[Z_i] \leq (1 - p)e^{\varepsilon} + \delta.
\]
\end{enumerate}

\end{solution}

\vskip .2in
\item Suppose you are given a private dataset $X \in \mathcal{X}^n$, where $\mathcal{X} = \{1, \dots, d\}$. That is, the dataset $X$ consists of $n$ integers $x_1, \dots, x_n$ between 1 and $d$. We want to generate a sample $Y \in \{1, \dots, d\}$ that satisfies $\min_{i=1}^n x_i \leq Y \leq \max_{i=1}^n x_i$ with high probability, but of course with $\varepsilon$-DP guarantee.

\begin{enumerate}
    \item[(a)] Define a utility function $q : \mathcal{X}^n \times \mathcal{X} \rightarrow \mathbb{Z}$ as:
    \[
    q(X, y) = \min \left\{ |\{i : x_i \leq y\}|, \; |\{i : x_i \geq y\}| \right\}
    \]
    where $|\{\cdot\}|$ denotes the size of the set. Let $\text{OPT}(X) = \max_{y \in \mathcal{X}} q(X, y)$, then show that
    \[
    \text{OPT}(X) \geq \frac{n - 1}{2}.
    \]

    \item[(b)] We use the exponential mechanism with range $\mathcal{X}$ with utility function $q$. Find the sensitivity $\Delta$.

    \item[(c)] Describe an exponential mechanism that satisfies $\varepsilon$-differential privacy, and show that
    \[
    \Pr\left[q(X, Y) \geq \text{OPT}(X) - \frac{2}{\varepsilon} \log\left( \frac{d}{\beta} \right) \right] \geq 1 - \beta.
    \]
    You can use the utility guarantee result from the lecture note.

    \item[(d)] Specify $C_1$ and $C_2$ that the above exponential mechanism satisfies
    \[
    \Pr\left[ \min_{i=1}^n x_i \leq y \leq \max_{i=1}^n x_i \right] \geq 1 - \beta
    \]
    for
    \[
    n \geq \frac{C_1}{\varepsilon} \ln\left( \frac{d}{\beta} \right) + C_2.
    \]
    Here, $C_1$ and $C_2$ should be constants that are independent of $n$, $d$, $\beta$, and $\varepsilon$.\\
    (Hint) Consider the event $q(X, y) \geq 1$.
\end{enumerate}


%%%%%%%%%%%%%%%%%%%%%%%%%% Problem 4 Solution %%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{solution}
\vskip .2in
\textbf{Solution.}\\
\begin{enumerate}
    \item[(a)] Since we can choose $y$ to be the $\lfloor n/2 \rfloor$-th element of the sorted list, we have
    \[
    \text{OPT}(X) \geq \left\lfloor \frac{n}{2} \right\rfloor \geq \frac{n - 1}{2}.
    \]

    \item[(b)] We now compute the sensitivity $\Delta$ of the utility function $q$. Let $X$ and $X'$ be neighboring datasets differing in one element. Then for any $y \in \mathcal{X}$,
    \begin{align*}
        & \left| |\{i : x_i \le y\}| - |\{i : x_i' \le y\}| \right| \le 1, \\
        & \left| |\{i : x_i \ge y\}| - |\{i : x_i' \ge y\}| \right| \le 1.
    \end{align*}
    Thus,
    \[
    |q(X, y) - q(X', y)| \le 1,
    \]
    so $\Delta = 1$.
    
    \item[(c)] Check the lecture note. We have $\Delta = 1$, $t = \log(1/\beta)$.

    \item[(d)] Observe that $y$ lies between $\min_i x_i$ and $\max_i x_i$ if and only if $q(X, y) \geq 1$. From part (c), with probability at least $1 - \beta$, the output $y$ satisfies:
    \[
    q(X, y) \geq \text{OPT}(X) - \frac{2}{\varepsilon} \ln\left( \frac{|\mathcal{X}|}{\beta} \right).
    \]
    As noted earlier, $\text{OPT}(X) \geq \frac{n - 1}{2}$, so if
    \[
    \frac{n - 1}{2} - \frac{2}{\varepsilon} \ln\left( \frac{d}{\beta} \right) \geq 1,
    \]
    then $q(X, y) \geq 1$ and therefore $y$ lies between $\min_i x_i$ and $\max_i x_i$.\\
    Solving the inequality,
    \[
    \frac{n - 1}{2} \geq \frac{2}{\varepsilon} \ln\left( \frac{d}{\beta} \right) + 1
    \quad \Rightarrow \quad
    n \geq \frac{4}{\varepsilon} \ln\left( \frac{d}{\beta} \right) + 3.
    \]
    Hence, the desired guarantee holds if we set constants $C_1 = 4$ and $C_2 = 3$:
    \[
    n \geq \frac{4}{\varepsilon} \ln\left( \frac{d}{\beta} \right) + 3
    \quad \Rightarrow \quad
    \Pr\left[ \min_i x_i \leq y \leq \max_i x_i \right] \geq 1 - \beta.
    \]

    
\end{enumerate}

\end{solution}

\end{enumerate}
