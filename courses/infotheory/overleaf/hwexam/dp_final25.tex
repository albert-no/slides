\item {\bf Nonuniform Time Scale Diffusion} {(\it 30 points)}\\
For a data variable $X$, consider a diffusion model with latent variables $Z_t$,
where $t$ runs from $t=0$ (least noisy) to $t=1$ (most noisy).
The distribution of latent variable $Z_t$ conditioned on $X$ for $t\in[0, 1]$ is given by
\begin{align*}
    q(Z_t|X)\sim \mathcal{N}(\alpha_t X, \sigma_t^2).
\end{align*}
Furthermore, for $t>s$, the conditional distribution $Z_t$ given $Z_s$ is given by
\begin{align*}
    q(Z_t|Z_s) \sim \mathcal{N}(\alpha_{t|s} Z_s, \sigma_{t|s}^2)
\end{align*}
where $\alpha_{t|s} = \alpha_t/\alpha_s$ and $\sigma_{t|s}^2 = \sigma_t^2 - \alpha_{t|s}^2\sigma_s^2$.
\begin{enumerate}
\item Let $q(Z_s|Z_t, X) \sim \mathcal{N}(\mu_Q(Z_t, X;s, t), \sigma_Q^2(s, t))$.
Find $\mu_Q(Z_t, X;s, t)$ and $\sigma_Q^2(s, t)$.  {(\it 10 points)}
\item Let $t_i = i/T$ for $0\leq i\leq T$.
Consider a following reverse distribution $p$:
\begin{align*}
    p(X) = \int_{\mathbf{Z}} p(Z_1) p(X|Z_0) \prod_{i=1}^T p(Z_{t_{i-1}}|Z_{t_i})\, d\mathbf{Z}
\end{align*}
where $Z$ represents the whole $(Z_{t_0}, \dots, Z_{t_T})$.

Show the following variational bound
\begin{align*}
    -\log p(X) \leq D(q(Z_1|X)\| p(Z_1)) + \field{E}_{q(Z_0|X)}[-\log p(X|Z_0)] + \mathcal{L}(X)
\end{align*}
where
\begin{align*}
    \mathcal{L}(X) = \sum_{i=1}^T \field{E}_{q(Z_{t_i}|X)} \left[D(q(Z_{t_{i-1}}|Z_{t_i}, X)\| p(Z_{t_{i-1}}|Z_{t_i})\right].
\end{align*}
{(\it 10 points)}
\item Let $p(Z_{t_{i-1}}|Z_{t_i}) = q(Z_{t_{i-1}}|Z_{t_i}, X=\hat{x}_{\theta}(Z_{t_i}; t_i))$.
Show that 
\begin{align*}
    \mathcal{L}(X) = \frac{1}{2} \sum_{i=1}^T \field{E}_{q(Z_{t_i}|X)} \left[\left(\mbox{SNR}(t_{i-1}) - \mbox{SNR}(t_i)\right) \|X - \hat{x}_\theta(Z_{t_i};t_i)\|^2\right]
\end{align*}
where $\mbox{SNR}(t) = \alpha_t^2 / \sigma_t^2$.
{(\it 10 points)}
\end{enumerate}

\item {\bf Masked Diffusion} {\it (30 points)}\\
Consider a masked discrete diffusion model with absorbing matrix.
More precisely, we have $X_0\sim p_{data}$ and the forward process $X_1, X_2, \dots, X_n$ where
$X_i\sim p_i$ with $p_i = Q_\text{absorb} p_{i-1}$ for all $i$. 
Note that
\[
Q_\text{absorb} = \left[ \begin{matrix}
    1 - \epsilon & 0 & \cdots & 0 & 0 \\
    0 & 1 - \epsilon & \cdots & 0 & 0 \\
    \vdots & \vdots & \ddots & \vdots & \vdots \\
    0 & 0 & \cdots & 1 - \epsilon & 0 \\
    \epsilon & \epsilon & \cdots & \epsilon & 1
\end{matrix} \right]
\]
\begin{enumerate}
    \item Find $p(X_i=x|X_0=x)$ as a function of $i$ and $\epsilon$. {\it (10 points)}
    \item Let $Q^{(i)}$ be a transition matrix that satisfies $p_i = Q^{(i)} p_0$. Find $Q^{(i)}$. {\it (10 points)}
    \item We re-parametrize the forward process $\{Y_t\}_{t=0}^T$ so that the $Y_t$ is now has continuous time scale.
    More precisely, we have $Y_{i\epsilon} = X_i$ where $T = n\epsilon$.
    Let $q_t$ be the distribution of $Y_t$ (i.e., $q_{i\epsilon} = p_i$).
    We still have $q_{t+\epsilon} = Q_\text{absorb} q_t$.
    Find $Q_t$ that satisfies
    \begin{align*}
        \frac{1}{\epsilon} (q_{t+\epsilon} - q_t) = Q_t q_t.
    \end{align*}
     {\it (10 points)}
\end{enumerate}

\item {\bf Discrete Langevin} {\it (40 points)}\\
In the homework problem, we showed that the Langevin SDE 
\[
dX_t=\frac{1}{2}\nabla_{X_t} \log p(X_t)dt+dW_t.
\]
preserves the distribution $p$ for all $t$, if the initial distribution satisfies $p_0=p$.
However, this result can be significantly strengthened. Specifically, even when the initial distribution $p_0$ differs from $p$,
the distribution of $X_t\sim p_t$ will converge to the target distribution $p$ as $t\rightarrow \infty$.

In this problem, we examine this convergence behavior in a simple setting where the target distribution $p$ is Gaussian.
Consider the probability density function of the Gaussian distribution $\mathcal{N}(\mu, \sigma^2)$:
\begin{align*}
    p(x) = \frac{1}{\sqrt{2\pi \sigma^2}} \exp\left(-\frac{(x-\mu)^2}{2\sigma^2}\right).
\end{align*}
Let this Gaussian $p(x)$ be our target distribution.

\begin{enumerate}
    \item Compute the score function $\frac{\partial}{\partial x} \log p(x)$. {\it (10 points)}
    \item Consider the discretized Langevin SDE with a small $\epsilon$:
    \begin{align*}
        X_{i+1} = X_i + \epsilon \cdot \frac{\partial}{\partial x}\log p(X_i) +\sqrt{2\epsilon}Z_i
    \end{align*}
    where $Z_i\sim \mathcal{N}(0, 1)$ is an independent Gaussian noise.
    Suppose the initial distribution is also standard Gaussian $X_0 \sim \mathcal{N}(0, 1)$,
    which differes from the target distribution $p$.
    Show that $X_i$ is Gaussian for all $i$. {\it (10 points)}
    \item Our goal is to show that $p_{X_n}$ converges to $p$ as $n$ grows.
    Since $X_i$ is Gaussian, we can write $X_i\sim \mathcal{N}(\mu_i, \sigma_i^2)$.
    Derive the recursive relations for $\mu_{i+1}$ in terms of $\mu_i$, and $\sigma_{i+1}^2$ in terms of $\sigma_i^2$.
    Show that $\mu_n\rightarrow \mu$ as $n\rightarrow \infty$. {\it (10 points)}
    \item Using a suitable approximation, show that $\sigma_n^2\rightarrow \sigma^2$ as $n\rightarrow \infty$,
    assuming $\epsilon$ is small.
     {\it (10 points)}
\end{enumerate}
